    61→function CategoryBadge({ category }: { category: string }) {

    65→    </Badge>
    66→  )

    69→function TransactionTypeBadge({ type }: { type: string }) {
    70→  const cls =
    71→    type === 'Add' ? 'bg-emerald-100 text-emerald-700 border-emerald-200 hover:bg-emerald-100'
    72→      : type === 'Issue' ? 'bg-sky-100 text-sky-700 border-sky-200 hover:bg-sky-100'
    73→        : 'bg-amber-100 text-amber-700 border-amber-200 hover:bg-amber-100'
    74→  return <Badge variant="outline" className={cn('font-medium', cls)}>{type}</Badge>
    75→}

    77→function formatQty(n: number | null | undefined): string {
    78→  const v = Number(n ?? 0)
    79→  if (!isFinite(v)) return '0'
    80→  return Number.isInteger(v) ? String(v) : String(v)
    81→}

   119→export function MaterialsPage() {
   120→  const { user } = useAppStore()
   121→
   122→  const [activeTab, setActiveTab] = useState<'inventory' | 'transactions'>('inventory')
   123→
   124→  // Inventory filters
   125→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
   126→  const [lowStockOnly, setLowStockOnly] = useState(false)
   127→
   128→  // Transaction filters
   129→  const [txnMaterialFilter, setTxnMaterialFilter] = useState<string>('all')
   130→  const [txnProjectFilter, setTxnProjectFilter] = useState<string>('all')
   131→  const [txnTypeFilter, setTxnTypeFilter] = useState<string>('all')
   132→
   133→  // Material dialog state (add/edit)
   134→  const [materialDialog, setMaterialDialog] = useState<{ open: boolean; mode: 'add' | 'edit'; material?: any }>({
   135→    open: false, mode: 'add',
   136→  })
   137→  const [mForm, setMForm] = useState<MaterialFormState>(EMPTY_MATERIAL_FORM)
   138→  const [savingMaterial, setSavingMaterial] = useState(false)
   139→
   140→  // Transaction dialog state
   141→  const [txnDialog, setTxnDialog] = useState<{ open: boolean; material?: any; type?: 'Add' | 'Issue' | 'Return' }>({
   142→    open: false,
   143→  })
   144→  const [tForm, setTForm] = useState<TxnFormState>(EMPTY_TXN_FORM)
   145→  const [savingTxn, setSavingTxn] = useState(false)
   146→
   147→  // Delete confirmation
   148→  const [deleteId, setDeleteId] = useState<string | null>(null)
   149→
   150→  // Fetches (single-source-of-truth: no server filter; filter client-side)
   151→  const { data: materialsData, loading: loadingMaterials, refetch: refetchMaterials } = useFetch<any>('/api/materials')
   152→  const { data: txnData, loading: loadingTxns, refetch: refetchTxns } = useFetch<any>('/api/transactions')
   153→  const { data: projectsData } = useFetch<any>('/api/projects')
   154→
   155→  const materials = (materialsData?.materials || []) as any[]
   156→  const transactions = (txnData?.transactions || []) as any[]
   157→  const projects = (projectsData?.projects || []) as any[]
   158→
   159→  // Filtered inventory
   160→  const filteredMaterials = useMemo(() => {
   161→    let arr = [...materials]
   162→    if (categoryFilter !== 'all') arr = arr.filter((m) => m.category === categoryFilter)
   163→    if (lowStockOnly) arr = arr.filter((m) => Number(m.stockQty ?? 0) <= Number(m.minStockLevel ?? 0))
   164→    return arr
   165→  }, [materials, categoryFilter, lowStockOnly])
   166→
   167→  // Filtered transactions
   168→  const filteredTxns = useMemo(() => {
   169→    let arr = [...transactions]
   170→    if (txnMaterialFilter !== 'all') arr = arr.filter((t) => t.materialId === txnMaterialFilter)
   171→    if (txnProjectFilter !== 'all') arr = arr.filter((t) => t.projectId === txnProjectFilter)
   172→    if (txnTypeFilter !== 'all') arr = arr.filter((t) => t.type === txnTypeFilter)
   173→    return arr
   174→  }, [transactions, txnMaterialFilter, txnProjectFilter, txnTypeFilter])
   175→
   176→  // Stats
   177→  const stats = useMemo(() => {
   178→    const total = materials.length
   179→    const lowStock = materials.filter((m) => Number(m.stockQty ?? 0) <= Number(m.minStockLevel ?? 0)).length
   180→    const outOfStock = materials.filter((m) => Number(m.stockQty ?? 0) <= 0).length
   181→    const totalStockValue = materials.reduce((s, m) => s + (Number(m.stockQty ?? 0) * Number(m.unitPrice ?? 0)), 0)
   182→    const categories = new Set(materials.map((m) => m.category)).size
   183→    return { total, lowStock, outOfStock, totalStockValue, categories }
   184→  }, [materials])
   185→
   186→  // Low stock materials (for alert card)
   187→  const lowStockMaterials = useMemo(
   188→    () => materials.filter((m) => Number(m.stockQty ?? 0) <= Number(m.minStockLevel ?? 0)),
   189→    [materials],
   190→  )
   191→
   192→  // Usage chart: total issued qty per material in last 30 days
   193→  const usageChart = useMemo(() => {
   194→    const cutoff = new Date()
   195→    cutoff.setDate(cutoff.getDate() - 30)
   196→    const map: Record<string, { name: string; issued: number; added: number; returned: number }> = {}
   197→    for (const t of transactions) {
   198→      const d = new Date(t.date)
   199→      if (d < cutoff) continue
   200→      const key = t.materialId || 'unknown'
   201→      const mname = t.material?.name || 'Unknown'
   202→      if (!map[key]) map[key] = { name: mname, issued: 0, added: 0, returned: 0 }
   203→      if (t.type === 'Issue') map[key].issued += Number(t.qty ?? 0)
   204→      else if (t.type === 'Add') map[key].added += Number(t.qty ?? 0)
   205→      else if (t.type === 'Return') map[key].returned += Number(t.qty ?? 0)
   206→    }
   207→    return Object.values(map).sort((a, b) => b.issued - a.issued).slice(0, 10)
   208→  }, [transactions])
   209→
   210→  // ---------- Handlers ----------
   211→  const openAddMaterial = () => {
   212→    setMForm(EMPTY_MATERIAL_FORM)
   213→    setMaterialDialog({ open: true, mode: 'add' })
   214→  }
   215→
   216→  const openEditMaterial = (m: any) => {
   217→    setMForm({
   218→      name: m.name || '',
   219→      category: m.category || 'SolarPanels',
   220→      unit: m.unit || 'pcs',
   221→      stockQty: Number(m.stockQty) || 0,
   222→      minStockLevel: Number(m.minStockLevel) || 0,
   223→      unitPrice: Number(m.unitPrice) || 0,
   224→      supplier: m.supplier || '',
   225→    })
   226→    setMaterialDialog({ open: true, mode: 'edit', material: m })
   227→  }
   228→
   229→  const submitMaterial = async () => {
   230→    if (!mForm.name.trim()) {
   231→      toast.error('Material name is required')
   232→      return
   233→    }
   234→    if (!mForm.category) {
   235→      toast.error('Material category is required')
   236→      return
   237→    }
   238→    setSavingMaterial(true)
   239→    try {
   240→      const body = {
   241→        name: mForm.name.trim(),
   242→        category: mForm.category,
   243→        unit: mForm.unit || 'pcs',
   244→        stockQty: Number(mForm.stockQty) || 0,
   245→        minStockLevel: Number(mForm.minStockLevel) || 0,
   246→        unitPrice: Number(mForm.unitPrice) || 0,
   247→        supplier: mForm.supplier?.trim() || null,
   248→      }
   249→      if (materialDialog.mode === 'add') {
   250→        await apiPost('/api/materials', body)
   251→        toast.success(`Material "${body.name}" added`)
   252→      } else if (materialDialog.mode === 'edit' && materialDialog.material?.id) {
   253→        await apiPut(`/api/materials/${materialDialog.material.id}`, body)
   254→        toast.success(`Material "${body.name}" updated`)
   255→      }
   256→      setMaterialDialog({ open: false, mode: 'add' })
   257→      refetchMaterials()
   258→    } catch (e: any) {
   259→      toast.error(e.message || 'Failed to save material')
   260→    } finally {
   261→      setSavingMaterial(false)
   262→    }
   263→  }
   264→
   265→  const openTxnDialog = (opts?: { material?: any; type?: 'Add' | 'Issue' | 'Return' }) => {
   266→    setTForm({
   267→      materialId: opts?.material?.id || '',
   268→      projectId: 'none',
   269→      type: opts?.type || 'Add',
   270→      qty: 0,
   271→      remarks: '',
   272→    })
   273→    setTxnDialog({ open: true, material: opts?.material, type: opts?.type })
   274→  }
   275→
   276→  const submitTxn = async () => {
   277→    if (!tForm.materialId) {
   278→      toast.error('Select a material')
   279→      return
   280→    }
   281→    if (!tForm.qty || Number(tForm.qty) <= 0) {
   282→      toast.error('Quantity must be greater than 0')
   283→      return
   284→    }
   285→    setSavingTxn(true)
   286→    try {
   287→      const body = {
   288→        materialId: tForm.materialId,
   289→        projectId: tForm.projectId && tForm.projectId !== 'none' ? tForm.projectId : null,
   290→        type: tForm.type,
   291→        qty: Number(tForm.qty),
   292→        remarks: tForm.remarks?.trim() || null,
   293→        date: new Date().toISOString(),
   294→      }
   295→      await apiPost<any>('/api/transactions', body)
   296→      toast.success(`${tForm.type} ${body.qty} — stock updated`)
   297→      setTxnDialog({ open: false })
   298→      refetchMaterials()
   299→      refetchTxns()
   300→    } catch (e: any) {
   301→      toast.error(e.message || 'Failed to record transaction')
   302→    } finally {
   303→      setSavingTxn(false)
   304→    }
   305→  }
   306→
   307→  const confirmDelete = async () => {
   308→    if (!deleteId) return
   309→    try {
   310→      await apiDelete(`/api/materials/${deleteId}`)
   311→      toast.success('Material deleted')
   312→      refetchMaterials()
   313→    } catch (e: any) {
   314→      toast.error(e.message || 'Failed to delete material')
   315→    } finally {
   316→      setDeleteId(null)
   317→    }
   318→  }
   319→
   320→  // ---------- Render: Loading ----------
   321→  if (loadingMaterials && !materialsData) {
   322→    return (
   323→      <div className="space-y-6">
   324→        <SectionHeader
   325→          section="materials"
   326→          title="Material Management"
   327→          description="Track solar materials, stock and transactions"
   328→          icon={<Package className="h-6 w-6" />}
   329→        />
   330→        <div>
   331→          <SubSection section="materials" title="Inventory Summary" icon={<BarChart3 className="h-4 w-4" />} />
   332→          <div className="grid grid-cols-2 md:grid-cols-4 gap-3 sm:gap-4">
   333→            {Array.from({ length: 4 }).map((_, i) => (
   334→              <Card key={i} className="overflow-hidden border-border/60">
   335→                <CardContent className="p-4 h-28">
   336→                  <Skeleton className="h-full w-full" />
   337→                </CardContent>
   338→              </Card>
   339→            ))}
   340→          </div>
   341→        </div>
   342→        <Card className="overflow-hidden border-border/60">
   343→          <CardContent className="p-6">
   344→            <Skeleton className="h-72 w-full" />
   345→          </CardContent>
   346→        </Card>
   347→      </div>
   348→    )
   349→  }
   350→
   351→  return (
   352→    <div className="space-y-6">
   353→      <SectionHeader
   354→        section="materials"
   355→        title="Material Management"
   356→        description="Track solar materials, stock and transactions"
   357→        icon={<Package className="h-6 w-6" />}
   358→        actions={
   359→          <>
   360→            <Button onClick={openAddMaterial} className="bg-violet-600 hover:bg-violet-700 text-white">
   361→              <Plus className="h-4 w-4" /> <span className="hidden sm:inline">Add Material</span><span className="sm:hidden">Add</span>
   362→            </Button>
   363→            <Button variant="outline" onClick={() => openTxnDialog()} className="border-violet-200 text-violet-700 hover:bg-violet-50">
   364→              <ArrowLeftRight className="h-4 w-4" /> <span className="hidden sm:inline">New Transaction</span><span className="sm:hidden">Transaction</span>
   365→            </Button>
   366→          </>
   367→        }
   368→      />
   369→
   370→      {/* Stat cards — Inventory Summary */}
   371→      <div>
   372→        <SubSection
   373→          section="materials"
   374→          title="Inventory Summary"
   375→          description="Stock levels and inventory value at a glance"
   376→          icon={<BarChart3 className="h-4 w-4" />}
   377→        />
   378→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3 sm:gap-4">
   379→          <StatCard
   380→            section="materials"
   381→            title="Total Materials"
   382→            value={stats.total ?? 0}
   383→            subtitle={`${stats.categories ?? 0} categories`}
   384→            icon={<Boxes className="h-5 w-5" />}
   385→          />
   386→          <StatCard
   387→            section="safety"
   388→            title="Low Stock Items"
   389→            value={stats.lowStock ?? 0}
   390→            subtitle={`${stats.outOfStock ?? 0} out of stock`}
   391→            icon={<AlertTriangle className="h-5 w-5" />}
   392→          />
   393→          <StatCard
   394→            section="materials"
   395→            title="Total Stock Value"
   396→            value={formatCurrency(stats.totalStockValue)}
   397→            subtitle="Inventory worth"
   398→            icon={<DollarSign className="h-5 w-5" />}
   399→          />
   400→          <StatCard
   401→            section="materials"
   402→            title="Categories"
   403→            value={stats.categories ?? 0}
   404→            subtitle="Distinct types"
   405→            icon={<Layers className="h-5 w-5" />}
   406→          />
   407→        </div>
   408→      </div>
   409→
   410→      {/* Low stock alert card */}
   411→      {lowStockMaterials.length > 0 && (
   412→        <Card className="border-amber-300 bg-amber-50/60">
   413→          <CardHeader className="pb-3">
   414→            <CardTitle className="text-base font-semibold flex items-center gap-2 text-amber-800">
   415→              <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-amber-200 text-amber-800">
   416→                <AlertTriangle className="h-4 w-4" />
   417→              </span>
   418→              Low Stock Alert
   419→              <Badge className="bg-amber-200 text-amber-800 border-amber-300 ml-1">{lowStockMaterials.length}</Badge>
   420→            </CardTitle>
   421→            <CardDescription className="text-xs text-amber-800/80 pl-9">
   422→              {lowStockMaterials.length} item(s) at or below minimum stock level — please restock
   423→            </CardDescription>
   424→          </CardHeader>
   425→          <CardContent>
   426→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3 max-h-72 overflow-y-auto ayk-scrollbar pr-1">
   427→              {lowStockMaterials.map((m) => {
   428→                const out = Number(m.stockQty ?? 0) <= 0
   429→                return (
   430→                  <div
   431→                    key={m.id}
   432→                    className={cn(
   433→                      'flex items-center justify-between rounded-lg border bg-white px-3 py-2.5',
   434→                      out ? 'border-red-200' : 'border-amber-200',
   435→                    )}
   436→                  >
   437→                    <div className="min-w-0">
   438→                      <div className="text-sm font-medium text-slate-900 truncate">{m.name || 'Unnamed'}</div>
   439→                      <div className="text-xs text-slate-500">
   440→                        Min: {formatQty(m.minStockLevel)} {m.unit || ''}
   441→                      </div>
   442→                    </div>
   443→                    <Badge
   444→                      className={cn(
   445→                        'shrink-0',
   446→                        out
   447→                          ? 'bg-red-100 text-red-700 border-red-200'
   448→                          : 'bg-amber-100 text-amber-800 border-amber-200',
   449→                      )}
   450→                    >
   451→                      {formatQty(m.stockQty)} {m.unit || ''}
   452→                    </Badge>
   453→                  </div>
   454→                )
   455→              })}
   456→            </div>
   457→          </CardContent>
   458→        </Card>
   459→      )}
   460→
   461→      {/* Tabs */}
   462→      <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as 'inventory' | 'transactions')}>
   463→        <TabsList className="bg-violet-50">
   464→          <TabsTrigger value="inventory" className="data-[state=active]:bg-white data-[state=active]:text-violet-700">
   465→            <Package className="h-4 w-4" /> <span className="hidden sm:inline">Inventory</span>
   466→          </TabsTrigger>
   467→          <TabsTrigger value="transactions" className="data-[state=active]:bg-white data-[state=active]:text-violet-700">
   468→            <ArrowLeftRight className="h-4 w-4" /> <span className="hidden sm:inline">Transactions</span>
   469→          </TabsTrigger>
   470→        </TabsList>
   471→
   472→        {/* Inventory tab */}
   473→        <TabsContent value="inventory" className="space-y-4">
   474→          {/* Filter bar */}
   475→          <Card className="shadow-sm border-border/60">
   476→            <CardContent className="p-4">
   477→              <div className="flex flex-col sm:flex-row sm:items-end gap-4">
   478→                <div className="flex-1 min-w-[180px]">
   479→                  <Label className="mb-1.5 text-xs text-muted-foreground uppercase tracking-wide">Category</Label>
   480→                  <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   481→                    <SelectTrigger className="w-full">
   482→                      <SelectValue placeholder="All categories" />
   483→                    </SelectTrigger>
   484→                    <SelectContent>
   485→                      <SelectItem value="all">All Categories</SelectItem>
   486→                      {MATERIAL_CATEGORIES.map((c) => (
   487→                        <SelectItem key={c} value={c}>
   488→                          {MATERIAL_CATEGORY_LABELS[c]}
   489→                        </SelectItem>
   490→                      ))}
   491→                    </SelectContent>
   492→                  </Select>
   493→                </div>
   494→                <div className="flex items-center gap-2 pb-1">
   495→                  <Checkbox
   496→                    id="low-stock-only"
   497→                    checked={lowStockOnly}
   498→                    onCheckedChange={(v) => setLowStockOnly(v === true)}
   499→                  />
   500→                  <Label htmlFor="low-stock-only" className="text-sm cursor-pointer">
   501→                    Low Stock Only
   502→                  </Label>
   503→                </div>
   504→                <div className="text-xs text-muted-foreground sm:ml-auto sm:pb-1">
   505→                  {filteredMaterials.length} of {materials.length} shown
   506→                </div>
   507→              </div>
   508→            </CardContent>
   509→          </Card>
   510→
   511→          {/* Inventory table */}
   512→          <Card className="shadow-sm border-border/60">
