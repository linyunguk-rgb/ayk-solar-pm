    61→/* ---- File-type icon (per doc.fileType) ----
    62→   pdf=red, image=sky, spreadsheet=emerald, word=blue, other=slate */
    63→function FileTypeIcon({ fileType, className }: { fileType: string; className?: string }) {
    64→  const t = (fileType || '').toLowerCase()
    65→  if (t.startsWith('image')) return <ImageIcon className={cn('h-7 w-7 text-sky-600', className)} />
    66→  if (t.includes('pdf')) return <FileText className={cn('h-7 w-7 text-red-500', className)} />
    67→  if (t.includes('sheet') || t.includes('excel') || t.includes('csv')) return <FileSpreadsheet className={cn('h-7 w-7 text-emerald-600', className)} />
    68→  if (t.includes('word') || t.includes('document')) return <FileText className={cn('h-7 w-7 text-blue-600', className)} />
    69→  return <File className={cn('h-7 w-7 text-slate-500', className)} />
    70→}
    71→
    72→function formatFileSize(bytes: number | undefined | null): string {
    73→  if (!bytes || bytes <= 0) return '—'
    74→  if (bytes < 1024) return `${bytes} B`
    75→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
    76→  if (bytes < 1024 * 1024 * 1024) return `${(bytes / (1024 * 1024)).toFixed(2)} MB`
    77→  return `${(bytes / (1024 * 1024 * 1024)).toFixed(2)} GB`
    78→}

   350→function PreviewBody({ doc }: { doc: any | null }) {
   351→  if (!doc) return null
   352→  const ft = (doc.fileType || '').toLowerCase()
   353→  if (ft.startsWith('image')) {
   354→    if (doc.url) {
   355→      return (
   356→        <div className="rounded-lg overflow-hidden border border-slate-200 bg-slate-50 flex items-center justify-center min-h-[260px]">
   357→          <img src={doc.url} alt={doc.name} className="max-h-[60vh] w-auto object-contain" />
   358→        </div>
   359→      )
   360→    }
   361→    return (
   362→      <div className="rounded-lg border border-dashed border-slate-300 bg-slate-50 p-10 text-center text-sm text-muted-foreground">
   363→        <ImageIcon className="h-10 w-10 mx-auto mb-2 text-slate-300" />
   364→        Preview not available for demo.
   365→      </div>
   366→    )
   367→  }
   368→  if (ft.includes('pdf')) {
   369→    return (
   370→      <div className="rounded-lg border border-slate-200 bg-slate-50 p-8 text-center text-sm text-muted-foreground">
   371→        <FileText className="h-10 w-10 mx-auto mb-2 text-red-400" />
   372→        <p className="font-medium text-slate-700 mb-1">PDF Document</p>
   373→        <p className="mb-3">In-browser PDF viewer is not available in this demo.</p>
   374→        <Button
   375→          variant="outline"
   376→          onClick={() => toast.info('Open in viewer (demo) — would launch PDF viewer in production')}
   377→        >
   378→          <Eye className="h-4 w-4 mr-2" /> Open in viewer
   379→        </Button>
   380→      </div>
   381→    )
   382→  }
   383→  return (
   384→    <div className="rounded-lg border border-slate-200 bg-slate-50 p-8 text-center text-sm text-muted-foreground">
   385→      <File className="h-10 w-10 mx-auto mb-2 text-slate-300" />
   386→      Preview not available for this file type.
   387→    </div>
   388→  )
   389→}

   471→function UploadDialog({

   474→  open: boolean

   477→  onUploaded: (name: string, objectUrl: string | null) => void
   478→}) {
   479→  const [name, setName] = useState('')
   480→  const [category, setCategory] = useState<string>('Drawings')
   481→  const [projectId, setProjectId] = useState<string>('')
   482→  const [description, setDescription] = useState('')
   483→  const [file, setFile] = useState<File | null>(null)
   484→  const [objectUrl, setObjectUrl] = useState<string | null>(null)
   485→  const [progress, setProgress] = useState(0)
   486→  const [uploading, setUploading] = useState(false)
   487→  const [dragOver, setDragOver] = useState(false)
   488→  const inputRef = useRef<HTMLInputElement>(null)
   489→
   490→  const reset = () => {
   491→    setName(''); setCategory('Drawings'); setProjectId(''); setDescription('')
   492→    setFile(null)
   493→    if (objectUrl) URL.revokeObjectURL(objectUrl)
   494→    setObjectUrl(null)
   495→    setProgress(0); setUploading(false); setDragOver(false)
   496→    if (inputRef.current) inputRef.current.value = ''
   497→  }
   498→
   499→  const handleFile = (f: File) => {
   500→    setFile(f)
   501→    if (!name) setName(f.name.replace(/\.[^.]+$/, ''))
   502→    // create object URL for preview if image
   503→    if (objectUrl) URL.revokeObjectURL(objectUrl)
   504→    const url = f.type.startsWith('image') ? URL.createObjectURL(f) : null
   505→    setObjectUrl(url)
   506→  }
   507→
   508→  const onDrop = (e: React.DragEvent) => {
   509→    e.preventDefault()
   510→    setDragOver(false)
   511→    const f = e.dataTransfer.files?.[0]
   512→    if (f) handleFile(f)
   513→  }
   514→
   515→  const handleSubmit = async () => {
   516→    if (!file) { toast.error('Please select a file'); return }
   517→    if (!category) { toast.error('Please select a category'); return }
   518→    setUploading(true)
   519→    setProgress(0)
   520→    // simulate progress
   521→    const step = setInterval(() => {
   522→      setProgress(p => {
   523→        if (p >= 90) { clearInterval(step); return 90 }
   524→        return p + 10
   525→      })
   526→    }, 80)
   527→    try {
   528→      await apiPost('/api/documents', {
   529→        name: name || file.name,
   530→        category,
   531→        projectId: projectId || null,
   532→        fileUrl: objectUrl || `/documents/${name || file.name}`,
   533→        fileType: file.type || 'unknown',
   534→        fileSize: file.size,
   535→        description,
   536→      })
   537→      setProgress(100)
   538→      clearInterval(step)
   539→      toast.success('Document uploaded')
   540→      onUploaded(name || file.name, objectUrl)
   541→      onOpenChange(false)
   542→      reset()
   543→    } catch (e: any) {
   544→      clearInterval(step)
   545→      toast.error(e.message || 'Failed to upload')
   546→    } finally {
   547→      setUploading(false)
   548→    }
   549→  }
   550→
   551→  return (
   552→    <Dialog open={open} onOpenChange={(o) => { onOpenChange(o); if (!o) reset() }}>
   553→      <DialogContent className="w-full sm:max-w-lg max-h-[90vh] overflow-y-auto ayk-scrollbar">
   554→        <DialogHeader>
   555→          <DialogTitle className="flex items-center gap-2">
   556→            <Upload className="h-4 w-4 text-teal-600" /> Upload Document
   557→          </DialogTitle>
   558→          <DialogDescription>Upload a project document, photo or report to the system.</DialogDescription>
   559→        </DialogHeader>
   560→
   561→        <div className="space-y-4">
   562→          {/* Drag-drop file area — large & prominent on mobile */}
   563→          <div
   564→            onDragOver={(e) => { e.preventDefault(); setDragOver(true) }}
   565→            onDragLeave={() => setDragOver(false)}
   566→            onDrop={onDrop}
   567→            onClick={() => inputRef.current?.click()}
   568→            className={cn(
   569→              'cursor-pointer rounded-xl border-2 border-dashed p-6 sm:p-8 text-center transition',
   570→              dragOver ? 'border-teal-400 bg-teal-50' : 'border-slate-300 hover:border-teal-400 hover:bg-slate-50'
   571→            )}
   572→          >
   573→            <input
   574→              ref={inputRef}
   575→              type="file"
   576→              className="hidden"
   577→              onChange={(e) => {
   578→                const f = e.target.files?.[0]
   579→                if (f) handleFile(f)
   580→              }}
   581→            />
   582→            {file ? (
   583→              <div className="flex items-center justify-center gap-3">
   584→                <FileTypeIcon fileType={file.type} />
   585→                <div className="text-left min-w-0">
   586→                  <div className="text-sm font-medium text-slate-900 truncate">{file.name}</div>
   587→                  <div className="text-xs text-muted-foreground">{formatFileSize(file.size)} · {file.type || 'unknown'}</div>
   588→                </div>
   589→              </div>
   590→            ) : (
   591→              <>
   592→                <div className="flex h-12 w-12 mx-auto mb-2 items-center justify-center rounded-full bg-teal-50">
   593→                  <Upload className="h-6 w-6 text-teal-600" />
   594→                </div>
   595→                <p className="text-sm font-medium text-slate-700">Drag & drop a file here, or click to browse</p>
   596→                <p className="text-xs text-muted-foreground mt-1">Images, PDFs, documents up to ~10MB</p>
   597→              </>
   598→            )}
   599→          </div>
   600→
   601→          {uploading && (
   602→            <div className="space-y-1">
   603→              <div className="flex items-center justify-between text-xs">
   604→                <span className="text-muted-foreground">Uploading…</span>
   605→                <span className="font-semibold text-teal-700 tabular-nums">{progress}%</span>
   606→              </div>
   607→              <Progress value={progress} className="h-2" />
   608→            </div>
   609→          )}
   610→
   611→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   612→            <div className="space-y-1.5">
   613→              <Label className="text-xs">Document Name</Label>
   614→              <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="Auto-filled from filename" />
   615→            </div>
   616→            <div className="space-y-1.5">
   617→              <Label className="text-xs">Category</Label>
   618→              <Select value={category} onValueChange={setCategory}>
   619→                <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   620→                <SelectContent>
   621→                  {DOCUMENT_CATEGORIES.map(c => <SelectItem key={c} value={c}>{CATEGORY_LABELS[c]}</SelectItem>)}
   622→                </SelectContent>
   623→              </Select>
   624→            </div>
   625→            <div className="space-y-1.5 sm:col-span-2">
   626→              <Label className="text-xs">Project (optional)</Label>
   627→              <Select value={projectId || '__none__'} onValueChange={(v) => setProjectId(v === '__none__' ? '' : v)}>
   628→                <SelectTrigger className="w-full"><SelectValue placeholder="Select project (optional)" /></SelectTrigger>
   629→                <SelectContent>
   630→                  <SelectItem value="__none__">No project</SelectItem>
   631→                  {projects.map((p: any) => <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>)}
   632→                </SelectContent>
   633→              </Select>
   634→            </div>
   635→            <div className="space-y-1.5 sm:col-span-2">
   636→              <Label className="text-xs">Description (optional)</Label>
   637→              <Textarea rows={3} value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Short description of the document…" />
   638→            </div>
   639→          </div>
   640→        </div>
   641→
   642→        <DialogFooter>
   643→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   644→          <Button onClick={handleSubmit} disabled={uploading} className="bg-teal-600 hover:bg-teal-700 text-white">
   645→            {uploading ? 'Uploading…' : (
   646→              <>
   647→                <Upload className="h-4 w-4 mr-2" /> Upload
   648→              </>
   649→            )}
   650→          </Button>
   651→        </DialogFooter>
   652→      </DialogContent>
   653→    </Dialog>
   654→  )
   655→}
