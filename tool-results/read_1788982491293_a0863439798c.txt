    98→    98→export function ExpensesPage() {
    99→    99→  const { user } = useAppStore()
   100→   100→
   101→   101→  // Filters
   102→   102→  const [projectFilter, setProjectFilter] = useState<string>('all')
   103→   103→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
   104→   104→  const [approvalFilter, setApprovalFilter] = useState<string>('all')
   105→   105→
   106→   106→  // Dialog state
   107→   107→  const [expenseDialog, setExpenseDialog] = useState<{ open: boolean; mode: 'add' | 'edit'; expense?: any }>({
   108→   108→    open: false, mode: 'add',
   109→   109→  })
   110→   110→  const [eForm, setEForm] = useState<ExpenseFormState>(EMPTY_EXPENSE_FORM)
   111→   111→  const [saving, setSaving] = useState(false)
   112→   112→
   113→   113→  // Delete confirmation
   114→   114→  const [deleteId, setDeleteId] = useState<string | null>(null)
   115→   115→
   116→   116→  // Fetches
   117→   117→  const { data: expensesData, loading: loadingExpenses, refetch: refetchExpenses } = useFetch<any>('/api/expenses')
   118→   118→  const { data: projectsData, loading: loadingProjects } = useFetch<any>('/api/projects')
   119→   119→
   120→   120→  const expenses = (expensesData?.expenses || []) as any[]
   121→   121→  const projects = (projectsData?.projects || []) as any[]
   122→   122→
   123→   123→  // Filtered expenses
   124→   124→  const filteredExpenses = useMemo(() => {
   125→   125→    let arr = [...expenses]
   126→   126→    if (projectFilter !== 'all') arr = arr.filter((e) => e.projectId === projectFilter)
   127→   127→    if (categoryFilter !== 'all') arr = arr.filter((e) => e.category === categoryFilter)
   128→   128→    if (approvalFilter !== 'all') arr = arr.filter((e) => e.approvalStatus === approvalFilter)
   129→   129→    return arr
   130→   130→  }, [expenses, projectFilter, categoryFilter, approvalFilter])
   131→   131→
   132→   132→  // Stats
   133→   133→  const stats = useMemo(() => {
   134→   134→    const total = expenses.reduce((s, e) => s + Number(e.amount), 0)
   135→   135→    const approved = expenses.filter((e) => e.approvalStatus === 'Approved')
   136→   136→    const pending = expenses.filter((e) => e.approvalStatus === 'Pending')
   137→   137→    const rejected = expenses.filter((e) => e.approvalStatus === 'Rejected')
   138→   138→    return {
   139→   139→      total,
   140→   140→      approvedSum: approved.reduce((s, e) => s + Number(e.amount), 0),
   141→   141→      approvedCount: approved.length,
   142→   142→      pendingSum: pending.reduce((s, e) => s + Number(e.amount), 0),
   143→   143→      pendingCount: pending.length,
   144→   144→      rejectedSum: rejected.reduce((s, e) => s + Number(e.amount), 0),
   145→   145→      rejectedCount: rejected.length,
   146→   146→    }
   147→   147→  }, [expenses])
   148→   148→
   149→   149→  // Approved expense sums per project
   150→   150→  const approvedByProject = useMemo(() => {
   151→   151→    const map: Record<string, number> = {}
   152→   152→    for (const e of expenses) {
   153→   153→      if (e.approvalStatus !== 'Approved') continue
   154→   154→      if (!e.projectId) continue
   155→   155→      map[e.projectId] = (map[e.projectId] || 0) + Number(e.amount)
   156→   156→    }
   157→   157→    return map
   158→   158→  }, [expenses])
   159→   159→
   160→   160→  // Budget vs Actual chart data
   161→   161→  const budgetChart = useMemo(() => {
   162→   162→    return projects.map((p) => ({
   163→   163→      name: p.name?.length > 14 ? p.name.slice(0, 14) + '…' : p.name,
   164→   164→      budget: Number(p.budget) || 0,
   165→   165→      actual: (Number(p.actualCost) || 0) + (approvedByProject[p.id] || 0),
   166→   166→    }))
   167→   167→  }, [projects, approvedByProject])
   168→   168→
   169→   169→  // Expense by category
   170→   170→  const categoryChart = useMemo(() => {
   171→   171→    const map: Record<string, number> = {}
   172→   172→    for (const e of expenses) {
   173→   173→      if (e.approvalStatus === 'Rejected') continue
   174→   174→      map[e.category] = (map[e.category] || 0) + Number(e.amount)
   175→   175→    }
   176→   176→    return EXPENSE_CATEGORIES.filter((c) => (map[c] || 0) > 0).map((c) => ({
   177→   177→      name: c,
   178→   178→      value: map[c] || 0,
   179→   179→    }))
   180→   180→  }, [expenses])
   181→   181→
   182→   182→  // Project profitability
   183→   183→  const profitability = useMemo(() => {
   184→   184→    return projects.map((p) => {
   185→   185→      const budget = Number(p.budget) || 0
   186→   186→      const actualCost = Number(p.actualCost) || 0
   187→   187→      const expensesApproved = approvedByProject[p.id] || 0
   188→   188→      const totalSpent = actualCost + expensesApproved
   189→   189→      const profit = budget - totalSpent
   190→   190→      const marginPct = budget > 0 ? Math.round((profit / budget) * 1000) / 10 : 0
   191→   191→      const usedPct = budget > 0 ? Math.min(100, Math.round((totalSpent / budget) * 1000) / 10) : 0
   192→   192→      return {
   193→   193→        id: p.id,
   194→   194→        name: p.name,
   195→   195→        budget,
   196→   196→        actualCost,
   197→   197→        expensesApproved,
   198→   198→        totalSpent,
   199→   199→        profit,
   200→   200→        marginPct,
   201→   201→        usedPct,
   202→   202→        overBudget: totalSpent > budget,
   203→   203→      }
   204→   204→    })
   205→   205→  }, [projects, approvedByProject])
   206→   206→
   207→   207→  // ---------- Handlers ----------
   208→   208→  const openAddExpense = () => {
   209→   209→    setEForm(EMPTY_EXPENSE_FORM)
   210→   210→    setExpenseDialog({ open: true, mode: 'add' })
   211→   211→  }
   212→   212→
   213→   213→  const openEditExpense = (e: any) => {
   214→   214→    const d = e.date ? new Date(e.date) : new Date()
   215→   215→    const tz = d.getTimezoneOffset() * 60000
   216→   216→    setEForm({
   217→   217→      date: new Date(d.getTime() - tz).toISOString().slice(0, 10),
   218→   218→      projectId: e.projectId || 'none',
   219→   219→      category: e.category || 'Tools & Equipment',
   220→   220→      description: e.description || '',
   221→   221→      amount: Number(e.amount) || 0,
   222→   222→      paidBy: e.paidBy || '',
   223→   223→      approvalStatus: e.approvalStatus || 'Pending',
   224→   224→      receiptUrl: e.receiptUrl || '',
   225→   225→    })
   226→   226→    setExpenseDialog({ open: true, mode: 'edit', expense: e })
   227→   227→  }
   228→   228→
   229→   229→  const submitExpense = async () => {
   230→   230→    if (!eForm.description.trim()) {
   231→   231→      toast.error('Description is required')
   232→   232→      return
   233→   233→    }
   234→   234→    if (!eForm.amount || Number(eForm.amount) <= 0) {
   235→   235→      toast.error('Amount must be greater than 0')
   236→   236→      return
   237→   237→    }
   238→   238→    setSaving(true)
   239→   239→    try {
   240→   240→      const body = {
   241→   241→        date: eForm.date ? new Date(eForm.date).toISOString() : new Date().toISOString(),
   242→   242→        projectId: eForm.projectId && eForm.projectId !== 'none' ? eForm.projectId : null,
   243→   243→        category: eForm.category,
   244→   244→        description: eForm.description.trim(),
   245→   245→        amount: Number(eForm.amount),
   246→   246→        paidBy: eForm.paidBy?.trim() || null,
   247→   247→        approvalStatus: eForm.approvalStatus,
   248→   248→        receiptUrl: eForm.receiptUrl?.trim() || null,
   249→   249→      }
   250→   250→      if (expenseDialog.mode === 'add') {
   251→   251→        await apiPost('/api/expenses', body)
   252→   252→        toast.success('Expense recorded')
   253→   253→      } else if (expenseDialog.mode === 'edit' && expenseDialog.expense?.id) {
   254→   254→        await apiPut(`/api/expenses/${expenseDialog.expense.id}`, body)
   255→   255→        toast.success('Expense updated')
   256→   256→      }
   257→   257→      setExpenseDialog({ open: false, mode: 'add' })
   258→   258→      refetchExpenses()
   259→   259→    } catch (e: any) {
   260→   260→      toast.error(e.message || 'Failed to save expense')
   261→   261→    } finally {
   262→   262→      setSaving(false)
   263→   263→    }
   264→   264→  }
   265→   265→
   266→   266→  const setApproval = async (id: string, status: 'Approved' | 'Rejected') => {
   267→   267→    try {
   268→   268→      await apiPut(`/api/expenses/${id}`, { approvalStatus: status })
   269→   269→      toast.success(`Expense ${status.toLowerCase()}`)
   270→   270→      refetchExpenses()
   271→   271→    } catch (e: any) {
   272→   272→      toast.error(e.message || `Failed to ${status.toLowerCase()} expense`)
   273→   273→    }
   274→   274→  }
   275→   275→
   276→   276→  const confirmDelete = async () => {

   285→   285→      setDeleteId(null)

   294→   294→    }
   295→   295→  }
   296→   296→
   297→   297→  // ---------- Render ----------
   298→   298→  if (loadingExpenses && !expensesData) {
   299→   299→    return (
   300→   300→      <div className="space-y-6">
   301→   301→        <PageHeader
   302→   302→          title="Expense Management"
   303→   303→          description="Track project expenses and approvals"
   304→   304→          icon={<DollarSign className="h-5 w-5" />}
   305→   305→        />
   306→   306→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3 sm:gap-4">
   307→   307→          {Array.from({ length: 4 }).map((_, i) => (
   308→   308→            <Card key={i}><CardContent className="p-4 h-28"><Skeleton className="h-full w-full" /></CardContent></Card>
   309→   309→          ))}
   310→   310→        </div>
   311→   311→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   312→   312→          <Card><CardContent className="p-4"><Skeleton className="h-72 w-full" /></CardContent></Card>
   313→   313→          <Card><CardContent className="p-4"><Skeleton className="h-72 w-full" /></CardContent></Card>
   314→   314→        </div>
   315→   315→      </div>
   316→   316→    )
   317→   317→  }
   318→   318→
   319→   319→  return (
   320→   320→    <div className="space-y-6">
   321→   321→      <PageHeader
   322→   322→        title="Expense Management"
   323→   323→        description="Track project expenses and approvals"
   324→   324→        icon={<DollarSign className="h-5 w-5" />}
   325→   325→        actions={
   326→   326→          <Button onClick={openAddExpense} className="bg-emerald-600 hover:bg-emerald-700 text-white">

   411→   411→          <CardContent>
