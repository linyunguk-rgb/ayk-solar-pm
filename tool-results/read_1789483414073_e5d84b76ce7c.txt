    99→function ProfileTab() {
   100→  const { user, setUser } = useAppStore()

   104→  const [saving, setSaving] = useState(false)
   105→
   106→  if (!user) return null
   107→
   108→  const handleSave = async () => {
   109→    if (!user) return
   110→    setSaving(true)
   111→    try {
   112→      const body: any = { name, phone }
   113→      if (password) body.password = password
   114→      const res = await apiPut<{ user: UserRow }>(`/api/users/${user.id}`, body)

   116→        ...user,
   117→        name: res.user.name,
   118→        phone: res.user.phone,
   119→      })
   120→      setPassword('')
   121→      toast.success('Profile updated')
   122→    } catch (e: any) {
   123→      toast.error(e.message || 'Failed to update profile')
   124→    } finally {
   125→      setSaving(false)
   126→    }
   127→  }
   128→
   129→  return (
   130→    <div className="space-y-4">
   131→      <SubSection
   132→        section="settings"

   218→function UsersTab() {
   219→  const { user } = useAppStore()
   220→  const { data, loading, refetch } = useFetch<{ users: UserRow[] }>('/api/users')
   221→  const [addOpen, setAddOpen] = useState(false)
   222→  const [editUser, setEditUser] = useState<UserRow | null>(null)
   223→  const [deleteUser, setDeleteUser] = useState<UserRow | null>(null)
   224→
   225→  if (!user) return null
   226→  if (user.role !== 'Admin') {
   227→    return (
   228→      <Card className="border-border/60 shadow-sm">
   229→        <CardContent className="py-4">
   230→          <EmptyState
   231→            icon={<Lock className="h-6 w-6" />}
   232→            title="Admins only"
   233→            description="User management is restricted to administrators. Contact your AYK administrator if you need access."
   234→          />
   235→        </CardContent>
   236→      </Card>
   237→    )
   238→  }
   239→
   240→  const users: UserRow[] = data?.users || []
   241→  const total = users.length
   242→  const active = users.filter((u) => u.isActive).length
   243→  const admins = users.filter((u) => u.role === 'Admin').length
   244→  const last30 = users.filter((u) => {
   245→    const d = new Date(u.createdAt)
   246→    if (isNaN(d.getTime())) return false
   247→    return (Date.now() - d.getTime()) < 30 * 24 * 60 * 60 * 1000
   248→  }).length
   249→
   250→  return (
   251→    <div className="space-y-4">
   252→      <SubSection
   253→        section="overview"
   254→        title="User Management"
   255→        description={`${total} registered user(s) — admin tools`}
   256→        icon={<Users className="h-4 w-4" />}
   257→        action={
   258→          <Dialog open={addOpen} onOpenChange={setAddOpen}>
   259→            <DialogTrigger asChild>
   260→              <Button className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2 w-full sm:w-auto">
   261→                <UserPlus className="h-4 w-4" /> Add User
   262→              </Button>
   263→            </DialogTrigger>
   264→            <DialogContent className="w-full sm:max-w-lg max-h-[90vh] overflow-y-auto ayk-scrollbar">
   265→              <AddUserForm
   266→                onClose={() => setAddOpen(false)}
   267→                onCreated={() => { refetch(); setAddOpen(false) }}
   268→              />
   269→            </DialogContent>
   270→          </Dialog>
   271→        }
   272→      />
   273→
   274→      {/* Stat cards for the users tab — emerald (overview) accent per spec */}
   275→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   276→        <StatCard title="Total Users" value={total} subtitle="All accounts" icon={<Users className="h-5 w-5" />} section="overview" compact />
   277→        <StatCard title="Active" value={active} subtitle="Currently enabled" icon={<UserCheck className="h-5 w-5" />} section="overview" compact />
   278→        <StatCard title="Admins" value={admins} subtitle="Full access" icon={<ShieldCheck className="h-5 w-5" />} section="overview" compact />
   279→        <StatCard title="New (30d)" value={last30} subtitle="Last 30 days" icon={<History className="h-5 w-5" />} section="overview" compact />
   280→      </div>
   281→
   282→      <Card className="border-border/60 shadow-sm">
   283→        <CardHeader className="pb-3">
   284→          <CardTitle className="text-base flex items-center gap-2">
   285→            <Users className="h-4 w-4 text-emerald-600" /> Registered Users
   286→          </CardTitle>
   287→          <CardDescription className="text-xs">{users.length} account(s) on file</CardDescription>
   288→        </CardHeader>
   289→        <CardContent className="pt-2">
   290→          {loading ? (
   291→            <LoadingState label="Loading users…" />
   292→          ) : users.length === 0 ? (
   293→            <EmptyState
   294→              icon={<Users className="h-6 w-6" />}
   295→              title="No users yet"
   296→              description="Add your first user to get started."
   297→            />

   299→            <ScrollArea className="max-h-[500px] ayk-scrollbar">
   300→              <Table>
   301→                <TableHeader>
   302→                  <TableRow className="sticky top-0 bg-card z-10">
   303→                    <TableHead className="min-w-[160px]">Name</TableHead>
   304→                    <TableHead className="min-w-[200px]">Email</TableHead>
   305→                    <TableHead className="min-w-[140px]">Role</TableHead>
   306→                    <TableHead className="min-w-[140px]">Phone</TableHead>
   307→                    <TableHead className="min-w-[100px]">Status</TableHead>

   309→                    <TableHead className="text-right min-w-[120px]">Actions</TableHead>
   310→                  </TableRow>
   311→                </TableHeader>
   312→                <TableBody>
   313→                  {users.map((u) => (
   314→                    <TableRow key={u.id} className="hover:bg-slate-50">
   315→                      <TableCell className="font-medium">{u.name}</TableCell>
   316→                      <TableCell className="text-muted-foreground text-xs">{u.email}</TableCell>
   317→                      <TableCell>
   318→                        <Badge className={`border ${ROLE_BADGE[u.role] || 'bg-slate-100 text-slate-700 border-slate-200'}`}>
   319→                          {ROLES[u.role as RoleKey] || u.role}
   320→                        </Badge>
   321→                      </TableCell>
   322→                      <TableCell className="text-xs">{u.phone || '—'}</TableCell>
   323→                      <TableCell>
   324→                        <Badge className={u.isActive ? 'bg-emerald-100 text-emerald-700 border-emerald-200 border' : 'bg-slate-100 text-slate-500 border-slate-200 border'}>

   335→                            variant="ghost"
   336→                            size="icon"

   340→                            title="Deactivate"

   353→        </CardContent>

   355→
   356→      {/* Edit dialog */}

   395→              Yes, deactivate

   404→function AddUserForm({ onClose, onCreated }: { onClose: () => void; onCreated: () => void }) {

   407→  const [password, setPassword] = useState('')

   409→  const [phone, setPhone] = useState('')
   410→  const [saving, setSaving] = useState(false)
   411→
   412→  const handleSubmit = async () => {
   413→    if (!email || !name || !password || !role) {
   414→      toast.error('Email, name, password and role are required')
   415→      return
   416→    }
   417→    setSaving(true)
   418→    try {
   419→      await apiPost('/api/users', { email, name, password, role, phone })
   420→      toast.success('User created')
   421→      onCreated()
   422→    } catch (e: any) {
   423→      toast.error(e.message || 'Failed to create user')

   425→      setSaving(false)
   426→    }
