    80→function ProfileTab() {

   508→function loadCompany(): CompanySettings {

   609→function loadNotif(): Record<string, boolean> {
   610→  const defaults: Record<string, boolean> = {
   611→    DelayedProject: true,
   612→    LowStock: true,
   613→    OverdueTask: true,
   614→    SafetyIncident: true,
   615→    PendingApproval: false,
   616→    DailyProgress: true,
   617→  }
   618→  if (typeof window === 'undefined') return defaults
   619→  try {
   620→    const raw = window.localStorage.getItem(NOTIF_LS_KEY)
   621→    if (raw) return { ...defaults, ...JSON.parse(raw) }
   622→  } catch { /* ignore */ }
   623→  return defaults
   624→}
   625→
   626→function NotificationsTab() {
   627→  const [toggles, setToggles] = useState<Record<string, boolean>>(() => loadNotif())
   628→
   629→  const updateToggle = useCallback((key: string, value: boolean) => {
   630→    setToggles((prev) => {
   631→      const next = { ...prev, [key]: value }
   632→      try { window.localStorage.setItem(NOTIF_LS_KEY, JSON.stringify(next)) } catch { /* ignore */ }
   633→      toast.success(`${NOTIFICATION_TYPES[key as keyof typeof NOTIFICATION_TYPES] || key} ${value ? 'enabled' : 'disabled'}`)
   634→      return next
   635→    })
   636→  }, [])
   637→
   638→  return (
   639→    <Card className="border-border/60 shadow-sm">
   640→      <CardHeader className="pb-3">
   641→        <CardTitle className="text-base flex items-center gap-2"><Bell className="h-4 w-4 text-emerald-600" /> Notification Preferences</CardTitle>
   642→        <CardDescription className="text-xs">Choose which alerts you want to receive (stored locally)</CardDescription>
   643→      </CardHeader>
   644→      <CardContent className="pt-2 divide-y divide-border/60">
   645→        {Object.entries(NOTIFICATION_TYPES).map(([key, label]) => (
   646→          <div key={key} className="flex items-center justify-between py-3 gap-3">
   647→            <div className="flex items-center gap-3">
   648→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-emerald-50 text-emerald-600">
   649→                <Bell className="h-4 w-4" />
   650→              </div>
   651→              <div>
   652→                <div className="text-sm font-medium text-foreground">{label}</div>
   653→                <div className="text-xs text-muted-foreground">Receive a notification when this event occurs</div>
   654→              </div>
   655→            </div>

   657→          </div>
