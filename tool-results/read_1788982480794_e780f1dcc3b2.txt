   489→
   490→function BoardView({
   491→  tasks, priorityFilter, onEdit, onDelete,
   492→}: {
   493→  tasks: TaskItem[]
   494→  priorityFilter: string
   495→  onEdit: (t: TaskItem) => void
   496→  onDelete: (id: string) => void
   497→}) {
   498→  const filtered = useMemo(() => {
   499→    if (priorityFilter === 'all') return tasks
   500→    return tasks.filter((t) => t.priority === priorityFilter)
   501→  }, [tasks, priorityFilter])
   502→
   503→  const grouped = useMemo(() => {
   504→    const g: Record<string, TaskItem[]> = { Todo: [], InProgress: [], Completed: [], Delayed: [] }
   505→    for (const t of filtered) {
   506→      const key = (t.status in g) ? t.status : 'Todo'
   507→      g[key].push(t)
   508→    }
   509→    return g
   510→  }, [filtered])
   511→
   512→  if (filtered.length === 0) {
   513→    return <EmptyState />
   514→  }
   515→
   516→  return (
   517→    <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-4 gap-4">
   518→      {COLUMN_DEFS.map((col) => (
   519→        <Card key={col.key} className="border-border/60 shadow-sm bg-slate-50/40">
   520→          <CardHeader className="pb-3 pt-4 px-4">
   521→            <div className="flex items-center justify-between">
   522→              <div className="flex items-center gap-2">
   523→                <span className={cn('h-2 w-2 rounded-full', col.dot)} />
   524→                <CardTitle className={cn('text-sm font-semibold', col.accent)}>{col.label}</CardTitle>
   525→              </div>
   526→              <Badge variant="secondary" className="bg-white text-slate-600 border border-slate-200">{grouped[col.key].length}</Badge>
   527→            </div>
   528→          </CardHeader>
   529→          <CardContent className="px-3 pb-3 space-y-2.5 max-h-[calc(100vh-22rem)] overflow-y-auto ayk-scrollbar">
   530→            {grouped[col.key].length === 0 ? (
   531→              <div className="text-center text-xs text-slate-400 py-8">No tasks</div>
   532→            ) : (
   533→              grouped[col.key].map((t) => (
   534→                <TaskCard key={t.id} task={t} onEdit={onEdit} onDelete={onDelete} />
   535→              ))
   536→            )}
   537→          </CardContent>
   538→        </Card>
   539→      ))}
   540→    </div>
   541→  )
   542→}

   603→function ListView({

   605→}: {

   607→  priorityFilter: string
   608→  onEdit: (t: TaskItem) => void
   609→  onDelete: (id: string) => void

   615→

   617→
   618→  return (
   619→    <Card className="border-border/60 shadow-sm">
   620→      <CardContent className="p-0">
   621→        <div className="max-h-[calc(100vh-18rem)] overflow-y-auto ayk-scrollbar">

   626→                <TableHead className="min-w-[140px]">Project</TableHead>
   627→                <TableHead>Assigned To</TableHead>
   628→                <TableHead>Priority</TableHead>
   629→                <TableHead>Status</TableHead>

   632→                <TableHead className="text-right">Actions</TableHead>

   634→            </TableHeader>

   638→                  <TableCell>
   639→                    <div className="flex items-center gap-2">
