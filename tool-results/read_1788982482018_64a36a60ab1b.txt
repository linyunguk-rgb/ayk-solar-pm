    70→function todayISO(): string {
    71→  const d = new Date()
    72→  const tz = d.getTimezoneOffset() * 60000
    73→  return new Date(d.getTime() - tz).toISOString().slice(0, 10)
    74→}

    89→  projectId: 'none',

    98→export function ExpensesPage() {
    99→  const { user } = useAppStore()
   100→
   101→  // Filters
   102→  const [projectFilter, setProjectFilter] = useState<string>('all')
   103→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
   104→  const [approvalFilter, setApprovalFilter] = useState<string>('all')
   105→
   106→  // Dialog state
   107→  const [expenseDialog, setExpenseDialog] = useState<{ open: boolean; mode: 'add' | 'edit'; expense?: any }>({
   108→    open: false, mode: 'add',
   109→  })
   110→  const [eForm, setEForm] = useState<ExpenseFormState>(EMPTY_EXPENSE_FORM)
   111→  const [saving, setSaving] = useState(false)
   112→
   113→  // Delete confirmation
   114→  const [deleteId, setDeleteId] = useState<string | null>(null)
   115→
   116→  // Fetches
   117→  const { data: expensesData, loading: loadingExpenses, refetch: refetchExpenses } = useFetch<any>('/api/expenses')
   118→  const { data: projectsData, loading: loadingProjects } = useFetch<any>('/api/projects')
   119→
   120→  const expenses = (expensesData?.expenses || []) as any[]
   121→  const projects = (projectsData?.projects || []) as any[]
   122→
   123→  // Filtered expenses
   124→  const filteredExpenses = useMemo(() => {
   125→    let arr = [...expenses]
   126→    if (projectFilter !== 'all') arr = arr.filter((e) => e.projectId === projectFilter)
   127→    if (categoryFilter !== 'all') arr = arr.filter((e) => e.category === categoryFilter)
   128→    if (approvalFilter !== 'all') arr = arr.filter((e) => e.approvalStatus === approvalFilter)
   129→    return arr
   130→  }, [expenses, projectFilter, categoryFilter, approvalFilter])
   131→
   132→  // Stats
   133→  const stats = useMemo(() => {
   134→    const total = expenses.reduce((s, e) => s + Number(e.amount), 0)
   135→    const approved = expenses.filter((e) => e.approvalStatus === 'Approved')
   136→    const pending = expenses.filter((e) => e.approvalStatus === 'Pending')
   137→    const rejected = expenses.filter((e) => e.approvalStatus === 'Rejected')
   138→    return {
   139→      total,
   140→      approvedSum: approved.reduce((s, e) => s + Number(e.amount), 0),
   141→      approvedCount: approved.length,
   142→      pendingSum: pending.reduce((s, e) => s + Number(e.amount), 0),
   143→      pendingCount: pending.length,
   144→      rejectedSum: rejected.reduce((s, e) => s + Number(e.amount), 0),
   145→      rejectedCount: rejected.length,
   146→    }
   147→  }, [expenses])
   148→
   149→  // Approved expense sums per project
   150→  const approvedByProject = useMemo(() => {
   151→    const map: Record<string, number> = {}
   152→    for (const e of expenses) {
   153→      if (e.approvalStatus !== 'Approved') continue
   154→      if (!e.projectId) continue
   155→      map[e.projectId] = (map[e.projectId] || 0) + Number(e.amount)
   156→    }
   157→    return map
   158→  }, [expenses])
   159→
   160→  // Budget vs Actual chart data
   161→  const budgetChart = useMemo(() => {
   162→    return projects.map((p) => ({
   163→      name: p.name?.length > 14 ? p.name.slice(0, 14) + '…' : p.name,
   164→      budget: Number(p.budget) || 0,
   165→      actual: (Number(p.actualCost) || 0) + (approvedByProject[p.id] || 0),
   166→    }))
   167→  }, [projects, approvedByProject])
   168→
   169→  // Expense by category
   170→  const categoryChart = useMemo(() => {
   171→    const map: Record<string, number> = {}
   172→    for (const e of expenses) {
   173→      if (e.approvalStatus === 'Rejected') continue
   174→      map[e.category] = (map[e.category] || 0) + Number(e.amount)
   175→    }
   176→    return EXPENSE_CATEGORIES.filter((c) => (map[c] || 0) > 0).map((c) => ({
   177→      name: c,
   178→      value: map[c] || 0,
   179→    }))
   180→  }, [expenses])
   181→
   182→  // Project profitability
   183→  const profitability = useMemo(() => {
   184→    return projects.map((p) => {
   185→      const budget = Number(p.budget) || 0
   186→      const actualCost = Number(p.actualCost) || 0
   187→      const expensesApproved = approvedByProject[p.id] || 0
   188→      const totalSpent = actualCost + expensesApproved
   189→      const profit = budget - totalSpent
   190→      const marginPct = budget > 0 ? Math.round((profit / budget) * 1000) / 10 : 0
   191→      const usedPct = budget > 0 ? Math.min(100, Math.round((totalSpent / budget) * 1000) / 10) : 0
   192→      return {
   193→        id: p.id,
   194→        name: p.name,
   195→        budget,
   196→        actualCost,
   197→        expensesApproved,
   198→        totalSpent,
   199→        profit,
   200→        marginPct,
   201→        usedPct,
   202→        overBudget: totalSpent > budget,
   203→      }
   204→    })
   205→  }, [projects, approvedByProject])
   206→
   207→  // ---------- Handlers ----------
   208→  const openAddExpense = () => {
   209→    setEForm(EMPTY_EXPENSE_FORM)
   210→    setExpenseDialog({ open: true, mode: 'add' })
   211→  }
   212→
   213→  const openEditExpense = (e: any) => {
   214→    const d = e.date ? new Date(e.date) : new Date()
   215→    const tz = d.getTimezoneOffset() * 60000
   216→    setEForm({
   217→      date: new Date(d.getTime() - tz).toISOString().slice(0, 10),
   218→      projectId: e.projectId || 'none',
   219→      category: e.category || 'Tools & Equipment',
   220→      description: e.description || '',
   221→      amount: Number(e.amount) || 0,
   222→      paidBy: e.paidBy || '',
   223→      approvalStatus: e.approvalStatus || 'Pending',
   224→      receiptUrl: e.receiptUrl || '',
   225→    })
   226→    setExpenseDialog({ open: true, mode: 'edit', expense: e })
   227→  }
   228→
   229→  const submitExpense = async () => {
   230→    if (!eForm.description.trim()) {
   231→      toast.error('Description is required')
   232→      return
   233→    }
   234→    if (!eForm.amount || Number(eForm.amount) <= 0) {
   235→      toast.error('Amount must be greater than 0')
   236→      return
   237→    }
   238→    setSaving(true)
   239→    try {
   240→      const body = {
   241→        date: eForm.date ? new Date(eForm.date).toISOString() : new Date().toISOString(),
   242→        projectId: eForm.projectId && eForm.projectId !== 'none' ? eForm.projectId : null,
   243→        category: eForm.category,
   244→        description: eForm.description.trim(),
   245→        amount: Number(eForm.amount),
   246→        paidBy: eForm.paidBy?.trim() || null,
   247→        approvalStatus: eForm.approvalStatus,
   248→        receiptUrl: eForm.receiptUrl?.trim() || null,
   249→      }
   250→      if (expenseDialog.mode === 'add') {
   251→        await apiPost('/api/expenses', body)
   252→        toast.success('Expense recorded')
   253→      } else if (expenseDialog.mode === 'edit' && expenseDialog.expense?.id) {
   254→        await apiPut(`/api/expenses/${expenseDialog.expense.id}`, body)
   255→        toast.success('Expense updated')
   256→      }
   257→      setExpenseDialog({ open: false, mode: 'add' })
   258→      refetchExpenses()
   259→    } catch (e: any) {
   260→      toast.error(e.message || 'Failed to save expense')
   261→    } finally {
   262→      setSaving(false)
   263→    }
   264→  }
   265→
   266→  const setApproval = async (id: string, status: 'Approved' | 'Rejected') => {
   267→    try {
   268→      await apiPut(`/api/expenses/${id}`, { approvalStatus: status })
   269→      toast.success(`Expense ${status.toLowerCase()}`)
   270→      refetchExpenses()
   271→    } catch (e: any) {
   272→      toast.error(e.message || `Failed to ${status.toLowerCase()} expense`)
   273→    }
   274→  }
   275→
   276→  const confirmDelete = async () => {
   277→    if (!deleteId) return
   278→    try {
   279→      await apiDelete(`/api/expenses/${deleteId}`)
   280→      toast.success('Expense deleted')
   281→      refetchExpenses()
   282→    } catch (e: any) {
   283→      toast.error(e.message || 'Failed to delete expense')
   284→    } finally {
   285→      setDeleteId(null)
   286→    }
   287→  }
   288→
   289→  const onReceiptUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
   290→    const file = e.target.files?.[0]
   291→    if (file) {
   292→      setEForm((s) => ({ ...s, receiptUrl: file.name }))
   293→      toast.success(`Receipt "${file.name}" attached`)
   294→    }
   295→  }
   296→
   297→  // ---------- Render ----------
   298→  if (loadingExpenses && !expensesData) {
   299→    return (
   300→      <div className="space-y-6">
   301→        <PageHeader
   302→          title="Expense Management"
   303→          description="Track project expenses and approvals"
   304→          icon={<DollarSign className="h-5 w-5" />}
   305→        />
   306→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3 sm:gap-4">
   307→          {Array.from({ length: 4 }).map((_, i) => (
   308→            <Card key={i}><CardContent className="p-4 h-28"><Skeleton className="h-full w-full" /></CardContent></Card>
   309→          ))}
   310→        </div>
   311→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   312→          <Card><CardContent className="p-4"><Skeleton className="h-72 w-full" /></CardContent></Card>
   313→          <Card><CardContent className="p-4"><Skeleton className="h-72 w-full" /></CardContent></Card>
   314→        </div>
   315→      </div>
   316→    )
   317→  }
   318→
   319→  return (
   320→    <div className="space-y-6">
   321→      <PageHeader
   322→        title="Expense Management"
   323→        description="Track project expenses and approvals"
   324→        icon={<DollarSign className="h-5 w-5" />}
   325→        actions={
   326→          <Button onClick={openAddExpense} className="bg-emerald-600 hover:bg-emerald-700 text-white">
   327→            <Plus className="h-4 w-4" /> Add Expense
   328→          </Button>
   329→        }
   330→      />
   331→
   332→      {/* Stat cards */}
   333→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3 sm:gap-4">
   334→        <StatCard
   335→          title="Total Expenses"
   336→          value={formatCurrency(stats.total)}
   337→          subtitle={`${expenses.length} entries`}
   338→          icon={<Wallet className="h-5 w-5" />}
   339→          accent="green"
   340→        />
   341→        <StatCard
   342→          title="Approved"
   343→          value={formatCurrency(stats.approvedSum)}
   344→          subtitle={`${stats.approvedCount} entries`}
   345→          icon={<CheckCircle2 className="h-5 w-5" />}
   346→          accent="blue"
   347→        />
   348→        <StatCard
   349→          title="Pending"
   350→          value={formatCurrency(stats.pendingSum)}
   351→          subtitle={`${stats.pendingCount} awaiting review`}
   352→          icon={<Clock className="h-5 w-5" />}
   353→          accent="orange"
   354→        />
   355→        <StatCard
   356→          title="Rejected"
   357→          value={formatCurrency(stats.rejectedSum)}
   358→          subtitle={`${stats.rejectedCount} entries`}
   359→          icon={<XCircle className="h-5 w-5" />}
   360→          accent="red"
   361→        />
   362→      </div>
   363→
   364→      {/* Charts row */}
   365→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   366→        <Card className="lg:col-span-2 shadow-sm border-border/60">
   367→          <CardHeader className="pb-2">
   368→            <CardTitle className="text-base font-semibold">Budget vs Actual</CardTitle>
   369→            <CardDescription className="text-xs">
   370→              Project budget vs (actual cost + approved expenses) in S$
   371→            </CardDescription>
   372→          </CardHeader>
   373→          <CardContent>
   374→            {budgetChart.length === 0 ? (
   375→              <div className="h-[260px] flex items-center justify-center text-sm text-muted-foreground">
   376→                No projects yet.
   377→              </div>
   378→            ) : (
   379→              <ResponsiveContainer width="100%" height={260}>
   380→                <BarChart data={budgetChart} margin={{ top: 8, right: 8, left: 0, bottom: 30 }}>
   381→                  <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" vertical={false} />
   382→                  <XAxis
   383→                    dataKey="name"
   384→                    tick={{ fontSize: 10, fill: '#64748b' }}
   385→                    interval={0}
   386→                    angle={-15}
   387→                    textAnchor="end"
   388→                    height={50}
   389→                  />
   390→                  <YAxis tick={{ fontSize: 11, fill: '#64748b' }} tickFormatter={(v) => formatCurrency(v)} />
   391→                  <RTooltip
   392→                    contentStyle={{ borderRadius: 8, border: '1px solid #e2e8f0', fontSize: 12 }}
   393→                    formatter={(v: any) => formatCurrency(v)}
   394→                  />
   395→                  <Legend iconType="circle" wrapperStyle={{ fontSize: 11 }} />
   396→                  <Bar dataKey="budget" name="Budget" fill="#10b981" radius={[4, 4, 0, 0]} />
   397→                  <Bar dataKey="actual" name="Actual + Expenses" fill="#f59e0b" radius={[4, 4, 0, 0]} />
   398→                </BarChart>
   399→              </ResponsiveContainer>
   400→            )}
   401→          </CardContent>
   402→        </Card>
   403→
   404→        <Card className="shadow-sm border-border/60">
   405→          <CardHeader className="pb-2">
   406→            <CardTitle className="text-base font-semibold">Expense by Category</CardTitle>
   407→            <CardDescription className="text-xs">
   408→              Total amount per category (excludes rejected)
   409→            </CardDescription>
   410→          </CardHeader>
   411→          <CardContent>
   412→            {categoryChart.length === 0 ? (
   413→              <div className="h-[260px] flex items-center justify-center text-sm text-muted-foreground">
   414→                No expenses yet.
   415→              </div>
   416→            ) : (
   417→              <ResponsiveContainer width="100%" height={260}>
   418→                <PieChart>
   419→                  <Pie
   420→                    data={categoryChart}
   421→                    dataKey="value"
   422→                    nameKey="name"
   423→                    cx="50%"
   424→                    cy="50%"
   425→                    innerRadius={50}
   426→                    outerRadius={85}
   427→                    paddingAngle={2}
   428→                  >
   429→                    {categoryChart.map((_, i) => (
   430→                      <Cell key={i} fill={PIE_COLORS[i % PIE_COLORS.length]} />
   431→                    ))}
   432→                  </Pie>
   433→                  <RTooltip
   434→                    contentStyle={{ borderRadius: 8, border: '1px solid #e2e8f0', fontSize: 12 }}
   435→                    formatter={(v: any) => formatCurrency(v)}
   436→                  />
   437→                  <Legend iconType="circle" wrapperStyle={{ fontSize: 10 }} />
   438→                </PieChart>
   439→              </ResponsiveContainer>
   440→            )}
   441→          </CardContent>
   442→        </Card>
   443→      </div>
   444→
   445→      {/* Project profitability */}
   446→      <Card className="shadow-sm border-border/60">
   447→        <CardHeader className="pb-3">
   448→          <CardTitle className="text-base font-semibold flex items-center gap-2">
   449→            <TrendingUp className="h-4 w-4 text-emerald-600" /> Project Profitability
   450→          </CardTitle>
   451→          <CardDescription className="text-xs">
   452→            Budget, total spent and profit margin per project
   453→          </CardDescription>
   454→        </CardHeader>
   455→        <CardContent>
   456→          {profitability.length === 0 ? (
   457→            <div className="py-6 text-center text-sm text-muted-foreground">No projects yet.</div>
   458→          ) : (
   459→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   460→              {profitability.map((p) => (
   461→                <div
   462→                  key={p.id}
   463→                  className={cn(
   464→                    'rounded-xl border p-4 transition-colors',
   465→                    p.overBudget
   466→                      ? 'border-red-200 bg-red-50/40'
   467→                      : 'border-slate-200 bg-slate-50/40',
   468→                  )}
   469→                >
   470→                  <div className="flex items-start justify-between gap-2 mb-3">
   471→                    <h4 className="font-semibold text-slate-900 text-sm truncate">{p.name}</h4>
   472→                    {p.overBudget ? (
   473→                      <Badge className="bg-red-100 text-red-700 border-red-200 shrink-0">Over Budget</Badge>
   474→                    ) : (
   475→                      <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200 shrink-0">On Track</Badge>
   476→                    )}
   477→                  </div>
   478→                  <div className="space-y-2 text-xs">
   479→                    <div className="flex items-center justify-between">
   480→                      <span className="text-slate-600">Budget</span>
   481→                      <span className="font-semibold text-slate-900 tabular-nums">{formatCurrency(p.budget)}</span>
   482→                    </div>
   483→                    <div className="flex items-center justify-between">
   484→                      <span className="text-slate-600">Actual Cost</span>
   485→                      <span className="font-medium text-slate-700 tabular-nums">{formatCurrency(p.actualCost)}</span>
   486→                    </div>
   487→                    <div className="flex items-center justify-between">
   488→                      <span className="text-slate-600">Approved Expenses</span>
   489→                      <span className="font-medium text-slate-700 tabular-nums">{formatCurrency(p.expensesApproved)}</span>
   490→                    </div>
   491→                    <div className="flex items-center justify-between border-t border-slate-200 pt-2 mt-1">
   492→                      <span className="text-slate-700 font-medium">Total Spent</span>
   493→                      <span className="font-semibold text-slate-900 tabular-nums">{formatCurrency(p.totalSpent)}</span>
   494→                    </div>
   495→                    <Progress value={p.usedPct} className="h-1.5" />
   496→                    <div className="flex items-center justify-between">
   497→                      <span className="text-slate-600">Used</span>
   498→                      <span className={cn(
   499→                        'font-medium tabular-nums',
   500→                        p.usedPct > 100 ? 'text-red-600' : 'text-slate-700',
   501→                      )}>
   502→                        {p.usedPct}%
   503→                      </span>
   504→                    </div>
   505→                    <div className="flex items-center justify-between border-t border-slate-200 pt-2 mt-1">
   506→                      <span className={cn('flex items-center gap-1 font-medium', p.profit >= 0 ? 'text-emerald-700' : 'text-red-700')}>
   507→                        {p.profit >= 0 ? <TrendingUp className="h-3 w-3" /> : <TrendingDown className="h-3 w-3" />}
   508→                        Profit
   509→                      </span>
   510→                      <span className={cn(
   511→                        'font-bold tabular-nums',
