   104→   104→export function ManpowerPage() {
   105→   105→  const { user } = useAppStore()
   106→   106→  const canManage = user?.role === 'Admin' || user?.role === 'ProjectManager'
   107→   107→
   108→   108→  // Filters
   109→   109→  const [projectFilter, setProjectFilter] = useState<string>('all')
   110→   110→  const [teamFilter, setTeamFilter] = useState<string>('all')
   111→   111→  const [statusFilter, setStatusFilter] = useState<string>('all')
   112→   112→
   113→   113→  const [tab, setTab] = useState<string>('workers')
   114→   114→
   115→   115→  const [dialogOpen, setDialogOpen] = useState(false)
   116→   116→  const [editing, setEditing] = useState<WorkerItem | null>(null)
   117→   117→  const [form, setForm] = useState<WorkerFormState>(emptyWorkerForm())
   118→   118→  const [deleteId, setDeleteId] = useState<string | null>(null)
   119→   119→  const [submitting, setSubmitting] = useState(false)
   120→   120→  const [actionLoading, setActionLoading] = useState<string | null>(null)
   121→   121→
   122→   122→  // 7-day window for attendance stats and productivity chart
   123→   123→  const attendanceWindow = useMemo(() => {
   124→   124→    const to = new Date(); to.setHours(23, 59, 59, 999)
   125→   125→    const from = new Date(); from.setDate(from.getDate() - 6); from.setHours(0, 0, 0, 0)
   126→   126→    return { from: from.toISOString(), to: to.toISOString() }
   127→   127→  }, [])
   128→   128→
   129→   129→  const workersUrl = useMemo(() => {
   130→   130→    const params = new URLSearchParams()
   131→   131→    if (projectFilter !== 'all') params.set('projectId', projectFilter)
   132→   132→    if (teamFilter !== 'all') params.set('team', teamFilter)
   133→   133→    if (statusFilter !== 'all') params.set('status', statusFilter)
   134→   134→    const q = params.toString()
   135→   135→    return `/api/workers${q ? `?${q}` : ''}`
   136→   136→  }, [projectFilter, teamFilter, statusFilter])
   137→   137→
   138→   138→  const { data, loading, refetch } = useFetch<{ workers: WorkerItem[] }>(workersUrl)
   139→   139→  const { data: projData } = useFetch<{ projects: ProjectOption[] }>('/api/projects')
   140→   140→  const { data: attData } = useFetch<{ records: AttendanceItem[] }>(
   141→   141→    `/api/attendance?from=${attendanceWindow.from}&to=${attendanceWindow.to}`
   142→   142→  )
   143→   143→
   144→   144→  const workers = data?.workers ?? []
   145→   145→  const projects = projData?.projects ?? []
   146→   146→  const attendanceRecords = attData?.records ?? []
   147→   147→
   148→   148→  // Teams derived from workers (unique, non-empty)
   149→   149→  const teams = useMemo(() => {
   150→   150→    const set = new Set<string>()
   151→   151→    for (const w of workers) if (w.team) set.add(w.team)
   152→   152→    return Array.from(set).sort()
   153→   153→  }, [workers])
   154→   154→
   155→   155→  // Stats
   156→   156→  const stats = useMemo(() => {
   157→   157→    const today = new Date()
   158→   158→    let onSite = 0
   159→   159→    let absent = 0
   160→   160→    let totalManHours = 0
   161→   161→
   162→   162→    for (const w of workers) {
   163→   163→      const todayAtt = (w.attendance || []).find((a) => isSameDay(new Date(a.date), today))
   164→   164→      if (todayAtt && todayAtt.status === 'Present') onSite++
   165→   165→      else if (w.status === 'Active') absent++
   166→   166→    }
   167→   167→    for (const r of attendanceRecords) totalManHours += Number(r.workingHours) || 0
   168→   168→    return {
   169→   169→      onSite,
   170→   170→      absent,
   171→   171→      total: workers.length,
   172→   172→      manHours: Math.round(totalManHours * 10) / 10,
   173→   173→    }
   174→   174→  }, [workers, attendanceRecords])
   175→   175→
   176→   176→  // Productivity chart data: man-hours per team over last 7 days
   177→   177→  const productivity = useMemo(() => {
   178→   178→    const teamHours: Record<string, number> = {}
   179→   179→    for (const r of attendanceRecords) {
   180→   180→      const team = r.worker?.team || 'Unassigned'
   181→   181→      teamHours[team] = (teamHours[team] || 0) + (Number(r.workingHours) || 0)
   182→   182→    }
   183→   183→    const entries = Object.entries(teamHours).map(([team, hours]) => ({
   184→   184→      team: team.length > 12 ? team.slice(0, 11) + '…' : team,
   185→   185→      hours: Math.round(hours * 10) / 10,
   186→   186→    }))
   187→   187→    entries.sort((a, b) => b.hours - a.hours)
   188→   188→    return entries.slice(0, 10)
   189→   189→  }, [attendanceRecords])
   190→   190→
   191→   191→  // Today's hours per worker (for table column)
   192→   192→  const todayHoursByWorker = useMemo(() => {
   193→   193→    const map: Record<string, number> = {}
   194→   194→    const now = new Date()
   195→   195→    for (const w of workers) {
   196→   196→      const todayAtt = (w.attendance || []).find((a) => isSameDay(new Date(a.date), now))
   197→   197→      if (!todayAtt) { map[w.id] = 0; continue }
   198→   198→      if (todayAtt.workingHours) {
   199→   199→        map[w.id] = Number(todayAtt.workingHours)
   200→   200→      } else if (todayAtt.checkIn) {
   201→   201→        const ms = now.getTime() - new Date(todayAtt.checkIn).getTime()
   202→   202→        map[w.id] = Math.max(0, Math.round((ms / (1000 * 60 * 60)) * 10) / 10)
   203→   203→      } else {
   204→   204→        map[w.id] = 0
   205→   205→      }
   206→   206→    }
   207→   207→    return map
   208→   208→  }, [workers])
   209→   209→
   210→   210→  const isWorkerCheckedIn = useCallback((w: WorkerItem) => {
   211→   211→    const today = new Date()
   212→   212→    const att = (w.attendance || []).find((a) => isSameDay(new Date(a.date), today))
   213→   213→    return !!(att && att.status === 'Present' && att.checkIn && !att.checkOut)
   214→   214→  }, [])
   215→   215→
   216→   216→  const isWorkerCheckedOut = useCallback((w: WorkerItem) => {
   217→   217→    const today = new Date()
   218→   218→    const att = (w.attendance || []).find((a) => isSameDay(new Date(a.date), today))
   219→   219→    return !!(att && att.checkIn && att.checkOut)
   220→   220→  }, [])
   221→   221→
   222→   222→  const handleCheckAction = async (worker: WorkerItem, action: 'checkin' | 'checkout') => {
   223→   223→    setActionLoading(worker.id)
   224→   224→    try {
   225→   225→      await apiPost(`/api/workers/${worker.id}`, { action })
   226→   226→      toast.success(action === 'checkin' ? `${worker.name} checked in` : `${worker.name} checked out`)
   227→   227→      await refetch()
   228→   228→    } catch (e: any) {
   229→   229→      toast.error(e.message || 'Failed to update attendance')
   230→   230→    } finally {
   231→   231→      setActionLoading(null)
   232→   232→    }
   233→   233→  }
   234→   234→
   235→   235→  const openNew = useCallback(() => {
   236→   236→    setEditing(null)
   237→   237→    setForm(emptyWorkerForm())
   238→   238→    setDialogOpen(true)
   239→   239→  }, [])
   240→   240→
   241→   241→  const openEdit = useCallback((w: WorkerItem) => {
   242→   242→    setEditing(w)
   243→   243→    setForm({
   244→   244→      name: w.name || '',
   245→   245→      employeeId: w.employeeId || '',
   246→   246→      role: w.role || 'Solar Installer',
   247→   247→      team: w.team || '',
   248→   248→      projectId: w.projectId || '',
   249→   249→      phone: w.phone || '',
   250→   250→      skillLevel: w.skillLevel || 'Intermediate',
   251→   251→      status: w.status || 'Active',
   252→   252→    })
   253→   253→    setDialogOpen(true)
   254→   254→  }, [])
   255→   255→
   256→   256→  const handleSubmit = async () => {
   257→   257→    if (!form.name.trim() || !form.employeeId.trim()) {
   258→   258→      toast.error('Name and Employee ID are required')
   259→   259→      return
   260→   260→    }
   261→   261→    setSubmitting(true)
   262→   262→    try {
   263→   263→      const payload = {
   264→   264→        name: form.name.trim(),
   265→   265→        employeeId: form.employeeId.trim(),
   266→   266→        role: form.role,
   267→   267→        team: form.team.trim() || undefined,
   268→   268→        projectId: form.projectId || undefined,
   269→   269→        phone: form.phone.trim() || undefined,
   270→   270→        skillLevel: form.skillLevel,
   271→   271→        status: form.status,
   272→   272→      }
   273→   273→      if (editing) {
   274→   274→        await apiPut(`/api/workers/${editing.id}`, payload)
   275→   275→        toast.success('Worker updated')
   276→   276→      } else {
   277→   277→        await apiPost('/api/workers', payload)
   278→   278→        toast.success('Worker added')
   279→   279→      }
   280→   280→      setDialogOpen(false)
   281→   281→      await refetch()
   282→   282→    } catch (e: any) {
   283→   283→      toast.error(e.message || 'Failed to save worker')
   284→   284→    } finally {
   285→   285→      setSubmitting(false)
   286→   286→    }
   287→   287→  }
   288→   288→
   289→   289→  const handleDelete = async () => {
   290→   290→    if (!deleteId) return
   291→   291→    try {
   292→   292→      await apiDelete(`/api/workers/${deleteId}`)
   293→   293→      toast.success('Worker removed')
   294→   294→      setDeleteId(null)
   295→   295→      await refetch()
   296→   296→    } catch (e: any) {
   297→   297→      toast.error(e.message || 'Failed to delete worker')
   298→   298→    }
   299→   299→  }
   300→   300→
   301→   301→  const clearFilters = () => {
   302→   302→    setProjectFilter('all'); setTeamFilter('all'); setStatusFilter('all')
   303→   303→  }
   304→   304→  const hasFilters = projectFilter !== 'all' || teamFilter !== 'all' || statusFilter !== 'all'
   305→   305→
   306→   306→  return (
   307→   307→    <div className="space-y-6">
   308→   308→      <PageHeader
   309→   309→        title="Manpower Management"
   310→   310→        description="Track workers, teams and attendance"
   311→   311→        icon={<Users className="h-5 w-5" />}
   312→   312→        actions={
   313→   313→          canManage ? (
   314→   314→            <Button onClick={openNew} className="bg-emerald-600 hover:bg-emerald-700 text-white">
   315→   315→              <Plus className="h-4 w-4 mr-2" /> Add Worker
   316→   316→            </Button>
   317→   317→          ) : undefined
   318→   318→        }
   319→   319→      />
   320→   320→
   321→   321→      {/* Stat cards */}
   322→   322→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3 sm:gap-4">
   323→   323→        <StatCard title="Workers On Site" value={stats.onSite} subtitle="Checked in today" icon={<UserCheck className="h-5 w-5" />} accent="green" />
   324→   324→        <StatCard title="Workers Absent" value={stats.absent} subtitle="Active, not present" icon={<UserX className="h-5 w-5" />} accent="red" />
   325→   325→        <StatCard title="Total Workers" value={stats.total} subtitle="In roster" icon={<Briefcase className="h-5 w-5" />} accent="blue" />
   326→   326→        <StatCard title="Total Man-hours" value={formatNumber(stats.manHours)} subtitle="Last 7 days" icon={<Clock className="h-5 w-5" />} accent="orange" />
   327→   327→      </div>
   328→   328→
   329→   329→      {/* Productivity chart */}
   330→   330→      <Card className="border-border/60 shadow-sm">
   331→   331→        <CardHeader className="pb-2">
   332→   332→          <div className="flex items-center justify-between">
   333→   333→            <div>
   334→   334→              <CardTitle className="text-base font-semibold flex items-center gap-2">
   335→   335→                <BarChart3 className="h-4 w-4 text-emerald-600" /> Team Productivity
   336→   336→              </CardTitle>
   337→   337→              <CardDescription className="text-xs">Man-hours per team — last 7 days</CardDescription>
   338→   338→            </div>
   339→   339→            <Badge variant="outline" className="border-slate-200 text-slate-600">{productivity.length} teams</Badge>
   340→   340→          </div>
   341→   341→        </CardHeader>
   342→   342→        <CardContent>
   343→   343→          {productivity.length === 0 ? (
   344→   344→            <div className="text-center text-sm text-slate-500 py-10">No attendance data for the past 7 days</div>
   345→   345→          ) : (
   346→   346→            <ResponsiveContainer width="100%" height={260}>
   347→   347→              <BarChart data={productivity} margin={{ top: 8, right: 8, left: -10, bottom: 0 }}>
   348→   348→                <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" vertical={false} />
   349→   349→                <XAxis dataKey="team" tick={{ fontSize: 11, fill: '#64748b' }} interval={0} angle={-15} textAnchor="end" height={50} />
   350→   350→                <YAxis tick={{ fontSize: 11, fill: '#64748b' }} />
   351→   351→                <RTooltip
   352→   352→                  contentStyle={{ borderRadius: 8, border: '1px solid #e2e8f0', fontSize: 12 }}
   353→   353→                  formatter={(v: any) => [`${v} h`, 'Man-hours']}
   354→   354→                />
   355→   355→                <Bar dataKey="hours" fill="#10b981" radius={[4, 4, 0, 0]} name="Man-hours" />
   356→   356→              </BarChart>
   357→   357→            </ResponsiveContainer>
   358→   358→          )}
   359→   359→        </CardContent>
   360→   360→      </Card>
   361→   361→
   362→   362→      {/* Tabs */}
   363→   363→      <Tabs value={tab} onValueChange={setTab}>
   364→   364→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   365→   365→          <TabsList>
   366→   366→            <TabsTrigger value="workers"><Users className="h-4 w-4 mr-1" /> Workers</TabsTrigger>
   367→   367→            <TabsTrigger value="attendance"><Clock className="h-4 w-4 mr-1" /> Attendance Log</TabsTrigger>
   368→   368→          </TabsList>
   369→   369→
   370→   370→          {/* Filter bar (shared) */}
   371→   371→          <div className="flex flex-wrap items-center gap-2">
   372→   372→            <Filter className="h-4 w-4 text-slate-400" />
   373→   373→            <Select value={projectFilter} onValueChange={setProjectFilter}>
   374→   374→              <SelectTrigger size="sm" className="w-[150px] bg-white"><SelectValue placeholder="All Projects" /></SelectTrigger>
   375→   375→              <SelectContent>
   376→   376→                <SelectItem value="all">All Projects</SelectItem>
   377→   377→                {projects.map((p) => <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>)}
   378→   378→              </SelectContent>
   379→   379→            </Select>
   380→   380→
   381→   381→            <Select value={teamFilter} onValueChange={setTeamFilter}>
   382→   382→              <SelectTrigger size="sm" className="w-[140px] bg-white"><SelectValue placeholder="All Teams" /></SelectTrigger>
   383→   383→              <SelectContent>
   384→   384→                <SelectItem value="all">All Teams</SelectItem>
   385→   385→                {teams.map((t) => <SelectItem key={t} value={t}>{t}</SelectItem>)}
   386→   386→              </SelectContent>
   387→   387→            </Select>
   388→   388→
   389→   389→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   390→   390→              <SelectTrigger size="sm" className="w-[130px] bg-white"><SelectValue placeholder="All Statuses" /></SelectTrigger>
   391→   391→              <SelectContent>
   392→   392→                <SelectItem value="all">All Statuses</SelectItem>
   393→   393→                {WORKER_STATUSES.map((s) => <SelectItem key={s} value={s}>{WORKER_STATUS_LABELS[s]}</SelectItem>)}
   394→   394→              </SelectContent>
   395→   395→            </Select>
   396→   396→
   397→   397→            {hasFilters && (
   398→   398→              <Button variant="ghost" size="sm" onClick={clearFilters} className="text-slate-500">
   399→   399→                <X className="h-3.5 w-3.5 mr-1" /> Clear
   400→   400→              </Button>
   401→   401→            )}
   402→   402→          </div>
   403→   403→        </div>
   404→   404→
   405→   405→        {/* Workers tab */}
   406→   406→        <TabsContent value="workers">
   407→   407→          {loading ? (
   408→   408→            <Card className="border-border/60 shadow-sm animate-pulse"><CardContent className="p-0 h-96" /></Card>
   409→   409→          ) : workers.length === 0 ? (
   410→   410→            <EmptyState message="No workers match your filters" />
   411→   411→          ) : (
   412→   412→            <Card className="border-border/60 shadow-sm">
   413→   413→              <CardContent className="p-0">
   414→   414→                <ScrollArea className="max-h-96">
   415→   415→                  <Table>
   416→   416→                    <TableHeader className="sticky top-0 z-10 bg-slate-50/95 backdrop-blur">
   417→   417→                      <TableRow className="hover:bg-transparent">
   418→   418→                        <TableHead>Emp ID</TableHead>
   419→   419→                        <TableHead>Name</TableHead>
   420→   420→                        <TableHead>Role</TableHead>
   421→   421→                        <TableHead>Team</TableHead>
   422→   422→                        <TableHead>Project</TableHead>
   423→   423→                        <TableHead>Skill</TableHead>
