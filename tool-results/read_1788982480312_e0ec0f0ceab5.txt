    48→// Distinct category badge colors (no indigo)
    49→const CATEGORY_BADGE: Record<string, string> = {
    50→  SolarPanels: 'bg-amber-100 text-amber-700 border-amber-200 hover:bg-amber-100',
    51→  DCCables: 'bg-sky-100 text-sky-700 border-sky-200 hover:bg-sky-100',

    56→  Bolts: 'bg-violet-100 text-violet-700 border-violet-200 hover:bg-violet-100',
    57→  Other: 'bg-slate-100 text-slate-700 border-slate-200 hover:bg-slate-100',

    60→function CategoryBadge({ category }: { category: string }) {
    61→  return (
    62→    <Badge variant="outline" className={cn('font-medium', CATEGORY_BADGE[category] || CATEGORY_BADGE.Other)}>
    63→      {MATERIAL_CATEGORY_LABELS[category] || category}
    64→    </Badge>
    65→  )
    66→}
    67→
    68→function TransactionTypeBadge({ type }: { type: string }) {
    69→  const cls =
    70→    type === 'Add' ? 'bg-emerald-100 text-emerald-700 border-emerald-200 hover:bg-emerald-100'
    71→      : type === 'Issue' ? 'bg-sky-100 text-sky-700 border-sky-200 hover:bg-sky-100'
    72→        : 'bg-amber-100 text-amber-700 border-amber-200 hover:bg-amber-100'
    73→  return <Badge variant="outline" className={cn('font-medium', cls)}>{type}</Badge>
    74→}

    76→function formatQty(n: number): string {
    77→  if (n === null || n === undefined) return '0'
    78→  return Number.isInteger(n) ? String(n) : String(n)

   117→export function MaterialsPage() {
   118→  const { user } = useAppStore()
   119→
   120→  const [activeTab, setActiveTab] = useState<'inventory' | 'transactions'>('inventory')
   121→
   122→  // Inventory filters
   123→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
   124→  const [lowStockOnly, setLowStockOnly] = useState(false)
   125→
   126→  // Transaction filters
   127→  const [txnMaterialFilter, setTxnMaterialFilter] = useState<string>('all')
   128→  const [txnProjectFilter, setTxnProjectFilter] = useState<string>('all')
   129→  const [txnTypeFilter, setTxnTypeFilter] = useState<string>('all')
   130→
   131→  // Material dialog state (add/edit)
   132→  const [materialDialog, setMaterialDialog] = useState<{ open: boolean; mode: 'add' | 'edit'; material?: any }>({
   133→    open: false, mode: 'add',
   134→  })
   135→  const [mForm, setMForm] = useState<MaterialFormState>(EMPTY_MATERIAL_FORM)
   136→  const [savingMaterial, setSavingMaterial] = useState(false)
   137→
   138→  // Transaction dialog state
   139→  const [txnDialog, setTxnDialog] = useState<{ open: boolean; material?: any; type?: 'Add' | 'Issue' | 'Return' }>({
   140→    open: false,
   141→  })
   142→  const [tForm, setTForm] = useState<TxnFormState>(EMPTY_TXN_FORM)
   143→  const [savingTxn, setSavingTxn] = useState(false)
   144→
   145→  // Delete confirmation
   146→  const [deleteId, setDeleteId] = useState<string | null>(null)
   147→
   148→  // Fetches (single-source-of-truth: no server filter; filter client-side)
   149→  const { data: materialsData, loading: loadingMaterials, refetch: refetchMaterials } = useFetch<any>('/api/materials')
   150→  const { data: txnData, loading: loadingTxns, refetch: refetchTxns } = useFetch<any>('/api/transactions')
   151→  const { data: projectsData } = useFetch<any>('/api/projects')
   152→
   153→  const materials = (materialsData?.materials || []) as any[]
   154→  const transactions = (txnData?.transactions || []) as any[]
   155→  const projects = (projectsData?.projects || []) as any[]
   156→
   157→  // Filtered inventory
   158→  const filteredMaterials = useMemo(() => {
   159→    let arr = [...materials]
   160→    if (categoryFilter !== 'all') arr = arr.filter((m) => m.category === categoryFilter)
   161→    if (lowStockOnly) arr = arr.filter((m) => Number(m.stockQty) <= Number(m.minStockLevel))
   162→    return arr
   163→  }, [materials, categoryFilter, lowStockOnly])
   164→
   165→  // Filtered transactions
   166→  const filteredTxns = useMemo(() => {
   167→    let arr = [...transactions]
   168→    if (txnMaterialFilter !== 'all') arr = arr.filter((t) => t.materialId === txnMaterialFilter)
   169→    if (txnProjectFilter !== 'all') arr = arr.filter((t) => t.projectId === txnProjectFilter)
   170→    if (txnTypeFilter !== 'all') arr = arr.filter((t) => t.type === txnTypeFilter)
   171→    return arr
   172→  }, [transactions, txnMaterialFilter, txnProjectFilter, txnTypeFilter])
   173→
   174→  // Stats
   175→  const stats = useMemo(() => {
   176→    const total = materials.length
   177→    const lowStock = materials.filter((m) => Number(m.stockQty) <= Number(m.minStockLevel)).length
   178→    const outOfStock = materials.filter((m) => Number(m.stockQty) <= 0).length
   179→    const totalStockValue = materials.reduce((s, m) => s + Number(m.stockQty) * Number(m.unitPrice), 0)
   180→    const categories = new Set(materials.map((m) => m.category)).size
   181→    return { total, lowStock, outOfStock, totalStockValue, categories }
   182→  }, [materials])
   183→
   184→  // Low stock materials (for alert card)
   185→  const lowStockMaterials = useMemo(
   186→    () => materials.filter((m) => Number(m.stockQty) <= Number(m.minStockLevel)),
   187→    [materials],
   188→  )
   189→
   190→  // Usage chart: total issued qty per material in last 30 days
   191→  const usageChart = useMemo(() => {
   192→    const cutoff = new Date()
   193→    cutoff.setDate(cutoff.getDate() - 30)
   194→    const map: Record<string, { name: string; issued: number; added: number; returned: number }> = {}
   195→    for (const t of transactions) {
   196→      const d = new Date(t.date)
   197→      if (d < cutoff) continue
   198→      const key = t.materialId || 'unknown'
   199→      const mname = t.material?.name || 'Unknown'
   200→      if (!map[key]) map[key] = { name: mname, issued: 0, added: 0, returned: 0 }
   201→      if (t.type === 'Issue') map[key].issued += Number(t.qty)
   202→      else if (t.type === 'Add') map[key].added += Number(t.qty)
   203→      else if (t.type === 'Return') map[key].returned += Number(t.qty)
   204→    }
   205→    return Object.values(map).sort((a, b) => b.issued - a.issued).slice(0, 10)
   206→  }, [transactions])
   207→
   208→  // ---------- Handlers ----------
   209→  const openAddMaterial = () => {
   210→    setMForm(EMPTY_MATERIAL_FORM)
   211→    setMaterialDialog({ open: true, mode: 'add' })
   212→  }
   213→
   214→  const openEditMaterial = (m: any) => {
   215→    setMForm({
   216→      name: m.name || '',
   217→      category: m.category || 'SolarPanels',
   218→      unit: m.unit || 'pcs',
   219→      stockQty: Number(m.stockQty) || 0,
   220→      minStockLevel: Number(m.minStockLevel) || 0,
   221→      unitPrice: Number(m.unitPrice) || 0,
   222→      supplier: m.supplier || '',
   223→    })
   224→    setMaterialDialog({ open: true, mode: 'edit', material: m })
   225→  }
   226→
   227→  const submitMaterial = async () => {
   228→    if (!mForm.name.trim()) {
   229→      toast.error('Material name is required')
   230→      return
   231→    }
   232→    if (!mForm.category) {
   233→      toast.error('Material category is required')
   234→      return
   235→    }
   236→    setSavingMaterial(true)
   237→    try {
   238→      const body = {
   239→        name: mForm.name.trim(),
   240→        category: mForm.category,
   241→        unit: mForm.unit || 'pcs',
   242→        stockQty: Number(mForm.stockQty) || 0,
   243→        minStockLevel: Number(mForm.minStockLevel) || 0,
   244→        unitPrice: Number(mForm.unitPrice) || 0,
   245→        supplier: mForm.supplier?.trim() || null,
   246→      }
   247→      if (materialDialog.mode === 'add') {
   248→        await apiPost('/api/materials', body)
   249→        toast.success(`Material "${body.name}" added`)
   250→      } else if (materialDialog.mode === 'edit' && materialDialog.material?.id) {
   251→        await apiPut(`/api/materials/${materialDialog.material.id}`, body)
   252→        toast.success(`Material "${body.name}" updated`)
   253→      }
   254→      setMaterialDialog({ open: false, mode: 'add' })
   255→      refetchMaterials()
   256→    } catch (e: any) {
   257→      toast.error(e.message || 'Failed to save material')
   258→    } finally {
   259→      setSavingMaterial(false)
   260→    }
   261→  }
   262→
   263→  const openTxnDialog = (opts?: { material?: any; type?: 'Add' | 'Issue' | 'Return' }) => {
   264→    setTForm({
   265→      materialId: opts?.material?.id || '',
   266→      projectId: 'none',
   267→      type: opts?.type || 'Add',
   268→      qty: 0,
   269→      remarks: '',
   270→    })
   271→    setTxnDialog({ open: true, material: opts?.material, type: opts?.type })
   272→  }
   273→
   274→  const submitTxn = async () => {
   275→    if (!tForm.materialId) {
   276→      toast.error('Select a material')
   277→      return
   278→    }
   279→    if (!tForm.qty || Number(tForm.qty) <= 0) {
   280→      toast.error('Quantity must be greater than 0')
   281→      return
   282→    }
   283→    setSavingTxn(true)
   284→    try {
   285→      const body = {
   286→        materialId: tForm.materialId,
   287→        projectId: tForm.projectId && tForm.projectId !== 'none' ? tForm.projectId : null,
   288→        type: tForm.type,
   289→        qty: Number(tForm.qty),
   290→        remarks: tForm.remarks?.trim() || null,
   291→        date: new Date().toISOString(),
   292→      }
   293→      const res = await apiPost<any>('/api/transactions', body)
   294→      toast.success(`${tForm.type} ${body.qty} ${tForm.materialId ? '' : ''} — stock updated`)
   295→      setTxnDialog({ open: false })
   296→      refetchMaterials()
   297→      refetchTxns()
   298→    } catch (e: any) {
   299→      toast.error(e.message || 'Failed to record transaction')
   300→    } finally {
   301→      setSavingTxn(false)
   302→    }
   303→  }
   304→
   305→  const confirmDelete = async () => {
   306→    if (!deleteId) return
   307→    try {
   308→      await apiDelete(`/api/materials/${deleteId}`)
   309→      toast.success('Material deleted')
   310→      refetchMaterials()
   311→    } catch (e: any) {
   312→      toast.error(e.message || 'Failed to delete material')
   313→    } finally {
   314→      setDeleteId(null)
   315→    }
   316→  }
   317→
   318→  // ---------- Render ----------
   319→  if (loadingMaterials && !materialsData) {
   320→    return (
   321→      <div className="space-y-6">
   322→        <PageHeader
   323→          title="Material Management"
   324→          description="Track solar materials, stock and transactions"
   325→          icon={<Package className="h-5 w-5" />}
   326→        />
   327→        <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3 sm:gap-4">
   328→          {Array.from({ length: 5 }).map((_, i) => (
   329→            <Card key={i}><CardContent className="p-4 h-28"><Skeleton className="h-full w-full" /></CardContent></Card>
   330→          ))}
   331→        </div>
   332→        <Card><CardContent className="p-6"><Skeleton className="h-72 w-full" /></CardContent></Card>
   333→      </div>
   334→    )
   335→  }
   336→
   337→  return (
   338→    <div className="space-y-6">
   339→      <PageHeader
   340→        title="Material Management"
   341→        description="Track solar materials, stock and transactions"
   342→        icon={<Package className="h-5 w-5" />}
   343→        actions={
   344→          <>
   345→            <Button onClick={openAddMaterial} className="bg-emerald-600 hover:bg-emerald-700 text-white">
   346→              <Plus className="h-4 w-4" /> Add Material
   347→            </Button>
   348→            <Button variant="outline" onClick={() => openTxnDialog()}>
   349→              <ArrowLeftRight className="h-4 w-4" /> New Transaction
   350→            </Button>
   351→          </>
   352→        }
   353→      />
   354→
   355→      {/* Stat cards */}
   356→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3 sm:gap-4">
   357→        <StatCard
   358→          title="Total Materials"
   359→          value={stats.total}
   360→          subtitle={`${stats.categories} categories`}
   361→          icon={<Boxes className="h-5 w-5" />}
   362→          accent="green"
   363→        />
   364→        <StatCard
   365→          title="Low Stock Items"
   366→          value={stats.lowStock}
   367→          subtitle="At/below minimum"
   368→          icon={<AlertTriangle className="h-5 w-5" />}
   369→          accent="red"
   370→        />
   371→        <StatCard
   372→          title="Total Stock Value"
   373→          value={formatCurrency(stats.totalStockValue)}
   374→          subtitle="Inventory worth"
   375→          icon={<DollarSign className="h-5 w-5" />}
   376→          accent="purple"
   377→        />
   378→        <StatCard
   379→          title="Categories"
   380→          value={stats.categories}
   381→          subtitle="Distinct types"
   382→          icon={<Layers className="h-5 w-5" />}
   383→          accent="blue"
   384→        />
   385→        <StatCard
   386→          title="Out of Stock"
   387→          value={stats.outOfStock}
   388→          subtitle="Zero on hand"
   389→          icon={<PackageX className="h-5 w-5" />}
   390→          accent="orange"
   391→        />
   392→      </div>
   393→
   394→      {/* Low stock alert card */}
   395→      {lowStockMaterials.length > 0 && (
   396→        <Card className="border-amber-200 bg-amber-50/40">
   397→          <CardHeader className="pb-3">
   398→            <CardTitle className="text-base font-semibold flex items-center gap-2 text-amber-700">
   399→              <AlertTriangle className="h-4 w-4" /> Low Stock Alert
   400→            </CardTitle>
   401→            <CardDescription className="text-xs text-amber-700/80">
   402→              {lowStockMaterials.length} item(s) at or below minimum stock level — please restock
   403→            </CardDescription>
   404→          </CardHeader>
   405→          <CardContent>
   406→            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3 max-h-64 overflow-y-auto ayk-scrollbar pr-1">
   407→              {lowStockMaterials.map((m) => (
   408→                <div
   409→                  key={m.id}
   410→                  className="flex items-center justify-between rounded-lg border border-amber-200 bg-white px-3 py-2.5"
   411→                >
   412→                  <div className="min-w-0">
   413→                    <div className="text-sm font-medium text-slate-900 truncate">{m.name}</div>
   414→                    <div className="text-xs text-slate-500">
   415→                      Min: {formatQty(m.minStockLevel)} {m.unit}
   416→                    </div>
   417→                  </div>
   418→                  <Badge className="bg-red-100 text-red-700 border-red-200 shrink-0">
   419→                    {formatQty(m.stockQty)} {m.unit}
   420→                  </Badge>
   421→                </div>
   422→              ))}
   423→            </div>
   424→          </CardContent>
   425→        </Card>
   426→      )}
   427→
   428→      {/* Tabs */}
   429→      <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as 'inventory' | 'transactions')}>
   430→        <TabsList className="bg-slate-100/70">
   431→          <TabsTrigger value="inventory"><Package className="h-4 w-4" /> Inventory</TabsTrigger>
   432→          <TabsTrigger value="transactions"><ArrowLeftRight className="h-4 w-4" /> Transactions</TabsTrigger>
   433→        </TabsList>
   434→
   435→        {/* Inventory tab */}
   436→        <TabsContent value="inventory" className="space-y-4">
   437→          {/* Filter bar */}
   438→          <Card className="shadow-sm border-border/60">
   439→            <CardContent className="p-4">
   440→              <div className="flex flex-col sm:flex-row sm:items-end gap-4">
   441→                <div className="flex-1 min-w-[180px]">
   442→                  <Label className="mb-1.5 text-xs text-muted-foreground uppercase tracking-wide">Category</Label>
   443→                  <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   444→                    <SelectTrigger className="w-full">
   445→                      <SelectValue placeholder="All categories" />
   446→                    </SelectTrigger>
   447→                    <SelectContent>
   448→                      <SelectItem value="all">All Categories</SelectItem>
   449→                      {MATERIAL_CATEGORIES.map((c) => (
   450→                        <SelectItem key={c} value={c}>
   451→                          {MATERIAL_CATEGORY_LABELS[c]}
   452→                        </SelectItem>
   453→                      ))}
   454→                    </SelectContent>
   455→                  </Select>
   456→                </div>
   457→                <div className="flex items-center gap-2 pb-1">
   458→                  <Checkbox
   459→                    id="low-stock-only"
   460→                    checked={lowStockOnly}
   461→                    onCheckedChange={(v) => setLowStockOnly(v === true)}
   462→                  />
   463→                  <Label htmlFor="low-stock-only" className="text-sm cursor-pointer">
   464→                    Low Stock Only
   465→                  </Label>
   466→                </div>
   467→                <div className="text-xs text-muted-foreground sm:ml-auto sm:pb-1">
   468→                  {filteredMaterials.length} of {materials.length} shown
   469→                </div>
   470→              </div>
   471→            </CardContent>
   472→          </Card>
   473→
   474→          {/* Inventory table */}
   475→          <Card className="shadow-sm border-border/60">
   476→            <CardHeader className="pb-3">
   477→              <CardTitle className="text-base font-semibold flex items-center gap-2">
   478→                <Warehouse className="h-4 w-4 text-emerald-600" /> Inventory List
   479→              </CardTitle>
   480→              <CardDescription className="text-xs">
   481→                Manage stock levels, pricing and supplier details
   482→              </CardDescription>
   483→            </CardHeader>
   484→            <CardContent className="px-0 pb-0">
   485→              <ScrollArea className="max-h-[500px] w-full">
   486→                <Table>
   487→                  <TableHeader className="sticky top-0 bg-card z-10">
   488→                    <TableRow>
   489→                      <TableHead>Name</TableHead>
   490→                      <TableHead>Category</TableHead>
   491→                      <TableHead className="text-right">Stock Qty</TableHead>
   492→                      <TableHead>Unit</TableHead>
   493→                      <TableHead className="text-right">Min Level</TableHead>
   494→                      <TableHead className="text-right">Unit Price</TableHead>
   495→                      <TableHead className="text-right">Stock Value</TableHead>
   496→                      <TableHead>Supplier</TableHead>
   497→                      <TableHead className="text-right">Actions</TableHead>
   498→                    </TableRow>
   499→                  </TableHeader>
   500→                  <TableBody>
   501→                    {filteredMaterials.length === 0 ? (
   502→                      <TableRow>
   503→                        <TableCell colSpan={9} className="text-center text-muted-foreground py-10">
   504→                          No materials match the current filters.
   505→                        </TableCell>
   506→                      </TableRow>
   507→                    ) : (
   508→                      filteredMaterials.map((m) => {
   509→                        const isLow = Number(m.stockQty) <= Number(m.minStockLevel)
   510→                        const stockValue = Number(m.stockQty) * Number(m.unitPrice)
   511→                        return (
   512→                          <TableRow key={m.id}>
   513→                            <TableCell className="font-medium text-slate-900">{m.name}</TableCell>
   514→                            <TableCell><CategoryBadge category={m.category} /></TableCell>
