    48→// ============================================================

    51→export function ProjectsPage() {
    52→  const { detailProjectId } = useAppStore()

    60→const EMPTY_FORM = {
    61→  name: '', location: '', client: '', capacity: '', totalPanels: '',
    62→  budget: '', status: 'Active', startDate: '', endDate: '',
    63→  description: '', managerId: '',
    64→}
    65→
    66→function ProjectsList() {
    67→  const { user, openProject } = useAppStore()
    68→  const canEdit = user?.role === 'Admin' || user?.role === 'ProjectManager'
    69→  const canDelete = user?.role === 'Admin'
    70→
    71→  const [statusFilter, setStatusFilter] = useState<string>('all')
    72→  const [search, setSearch] = useState('')
    73→  const [view, setView] = useState<'cards' | 'table'>('cards')
    74→
    75→  const listUrl = statusFilter === 'all' ? '/api/projects' : `/api/projects?status=${statusFilter}`
    76→  const { data, loading, refetch } = useFetch<any>(listUrl)
    77→
    78→  const { data: usersData } = useFetch<any>('/api/users')
    79→  const users = (usersData?.users || []).filter(
    80→    (u: any) => u.role === 'ProjectManager' || u.role === 'Admin' || u.role === 'Engineer'
    81→  )
    82→
    83→  const [dialogOpen, setDialogOpen] = useState(false)
    84→  const [editId, setEditId] = useState<string | null>(null)
    85→  const [form, setForm] = useState<any>({ ...EMPTY_FORM })
    86→  const [submitting, setSubmitting] = useState(false)
    87→  const [deleteId, setDeleteId] = useState<string | null>(null)
    88→
    89→  const projects = (data?.projects || []).filter((p: any) => {
    90→    if (!search) return true
    91→    const q = search.toLowerCase()
    92→    return (
    93→      p.name?.toLowerCase().includes(q) ||
    94→      p.location?.toLowerCase().includes(q) ||
    95→      p.code?.toLowerCase().includes(q)
    96→    )
    97→  })
    98→
    99→  function openCreate() {
   100→    setEditId(null)
   101→    setForm({ ...EMPTY_FORM })
   102→    setDialogOpen(true)
   103→  }
   104→
   105→  function openEdit(p: any) {
   106→    setEditId(p.id)
   107→    setForm({
   108→      name: p.name || '',
   109→      location: p.location || '',
   110→      client: p.client || '',
   111→      capacity: p.capacity || '',
   112→      totalPanels: String(p.totalPanels ?? ''),
   113→      budget: String(p.budget ?? ''),
   114→      status: p.status || 'Active',
   115→      startDate: p.startDate ? toInputDate(p.startDate) : '',
   116→      endDate: p.endDate ? toInputDate(p.endDate) : '',
   117→      description: p.description || '',
   118→      managerId: p.managerId || '',
   119→    })
   120→    setDialogOpen(true)
   121→  }
   122→
   123→  async function onSubmit() {
   124→    if (!form.name || !form.location || !form.startDate || !form.endDate) {
   125→      toast.error('Please fill in name, location, start date and end date.')
   126→      return
   127→    }
   128→    setSubmitting(true)
   129→    try {
   130→      const payload = {
   131→        ...form,
   132→        totalPanels: Number(form.totalPanels) || 0,
   133→        budget: Number(form.budget) || 0,
   134→      }
   135→      if (editId) {
   136→        await apiPut(`/api/projects/${editId}`, payload)
   137→        toast.success('Project updated')
   138→      } else {
   139→        await apiPost('/api/projects', payload)
   140→        toast.success('Project created')
   141→      }
   142→      setDialogOpen(false)
   143→      refetch()
   144→    } catch (e: any) {
   145→      toast.error(e.message || 'Failed to save project')
   146→    } finally {
   147→      setSubmitting(false)
   148→    }
   149→  }
   150→
   151→  async function onDelete() {
   152→    if (!deleteId) return
   153→    try {
   154→      await apiDelete(`/api/projects/${deleteId}`)
   155→      toast.success('Project deleted')
   156→      setDeleteId(null)
   157→      refetch()
   158→    } catch (e: any) {
   159→      toast.error(e.message || 'Failed to delete project')
   160→    }
   161→  }
   162→
   163→  return (
   164→    <div className="space-y-6">
   165→      <PageHeader
   166→        title="Projects"
   167→        description="Manage all solar construction projects"
   168→        icon={<FolderKanban className="h-5 w-5" />}
   169→        actions={
   170→          canEdit ? (
   171→            <Button onClick={openCreate} className="bg-emerald-600 hover:bg-emerald-700 text-white">
   172→              <Plus className="h-4 w-4" /> New Project
   173→            </Button>
   174→          ) : null
   175→        }
   176→      />
   177→
   178→      {/* Filter bar */}
   179→      <Card className="border-border/60 shadow-sm">
   180→        <CardContent className="p-4 flex flex-col sm:flex-row gap-3 sm:items-center">
   181→          <div className="relative flex-1 min-w-0">
   182→            <Search className="h-4 w-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   183→            <Input
   184→              placeholder="Search by name, location, code..."
   185→              value={search}
   186→              onChange={(e) => setSearch(e.target.value)}
   187→              className="pl-9"
   188→            />
   189→          </div>
   190→          <div className="flex items-center gap-3 flex-wrap">
   191→            <div className="flex items-center gap-2">
   192→              <Filter className="h-4 w-4 text-muted-foreground" />
   193→              <Select value={statusFilter} onValueChange={setStatusFilter}>
   194→                <SelectTrigger className="w-[150px]">
   195→                  <SelectValue placeholder="All statuses" />
   196→                </SelectTrigger>
   197→                <SelectContent>
   198→                  <SelectItem value="all">All Statuses</SelectItem>
   199→                  {PROJECT_STATUSES.map((s) => (
   200→                    <SelectItem key={s} value={s}>{PROJECT_STATUS_LABELS[s]}</SelectItem>
   201→                  ))}
   202→                </SelectContent>
   203→              </Select>
   204→            </div>
   205→            <div className="flex rounded-md border border-border/60 overflow-hidden">
   206→              <Button
   207→                size="sm"
   208→                variant={view === 'cards' ? 'default' : 'ghost'}
   209→                className={`rounded-none ${view === 'cards' ? 'bg-emerald-600 text-white hover:bg-emerald-700' : ''}`}
   210→                onClick={() => setView('cards')}
   211→              >
   212→                Cards
   213→              </Button>
   214→              <Button
   215→                size="sm"
   216→                variant={view === 'table' ? 'default' : 'ghost'}
   217→                className={`rounded-none ${view === 'table' ? 'bg-emerald-600 text-white hover:bg-emerald-700' : ''}`}
   218→                onClick={() => setView('table')}
   219→              >
   220→                Table
   221→              </Button>
   222→            </div>
   223→          </div>
   224→        </CardContent>
   225→      </Card>
   226→
   227→      {loading ? (
   228→        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
   229→          {Array.from({ length: 6 }).map((_, i) => (
   230→            <Card key={i} className="animate-pulse">
   231→              <CardContent className="p-5 h-48" />
   232→            </Card>
   233→          ))}
   234→        </div>
   235→      ) : projects.length === 0 ? (
   236→        <Card className="border-border/60 shadow-sm">
   237→          <CardContent className="p-12 flex flex-col items-center justify-center text-center">
   238→            <div className="h-12 w-12 rounded-full bg-muted flex items-center justify-center mb-3">
   239→              <FolderKanban className="h-6 w-6 text-muted-foreground" />
   240→            </div>
   241→            <h3 className="text-sm font-medium">No projects found</h3>
   242→            <p className="text-xs text-muted-foreground mt-1">
   243→              Try adjusting your filters or create a new project.
   244→            </p>
   245→            {canEdit && (
   246→              <Button onClick={openCreate} className="mt-4 bg-emerald-600 hover:bg-emerald-700 text-white">
   247→                <Plus className="h-4 w-4" /> New Project
   248→              </Button>
   249→            )}
   250→          </CardContent>
   251→        </Card>
   252→      ) : view === 'cards' ? (
   253→        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
   254→          {projects.map((p: any) => (
   255→            <ProjectCard key={p.id} project={p} onOpen={() => openProject(p.id)} />
   256→          ))}
   257→        </div>
   258→      ) : (
   259→        <Card className="border-border/60 shadow-sm">
   260→          <CardContent className="p-0">
   261→            <ScrollArea className="max-h-[600px] ayk-scrollbar">
   262→              <Table>
   263→                <TableHeader>
   264→                  <TableRow>
   265→                    <TableHead>Code</TableHead>
   266→                    <TableHead>Name</TableHead>
   267→                    <TableHead>Location</TableHead>
   268→                    <TableHead>Status</TableHead>
   269→                    <TableHead>Progress</TableHead>
   270→                    <TableHead>Panels</TableHead>
   271→                    <TableHead>Budget</TableHead>
   272→                    <TableHead>Manager</TableHead>
   273→                    <TableHead>End Date</TableHead>
   274→                    <TableHead className="text-right">Actions</TableHead>
   275→                  </TableRow>
   276→                </TableHeader>
   277→                <TableBody>
   278→                  {projects.map((p: any) => (
   279→                    <TableRow
   280→                      key={p.id}
   281→                      className="cursor-pointer hover:bg-muted/40"
   282→                      onClick={() => openProject(p.id)}
   283→                    >
   284→                      <TableCell className="font-mono text-xs">{p.code}</TableCell>
   285→                      <TableCell className="font-medium">{p.name}</TableCell>
   286→                      <TableCell className="text-muted-foreground">{p.location}</TableCell>
   287→                      <TableCell><StatusBadge status={p.status} /></TableCell>
   288→                      <TableCell>
   289→                        <div className="flex items-center gap-2 min-w-[100px]">
   290→                          <Progress value={p.overallProgress || 0} className="h-1.5" />
   291→                          <span className="text-xs font-medium">{p.overallProgress || 0}%</span>
   292→                        </div>
   293→                      </TableCell>
   294→                      <TableCell className="text-xs">
   295→                        {formatNumber(p.installedPanels)} / {formatNumber(p.totalPanels)}
   296→                      </TableCell>
   297→                      <TableCell className="text-xs">{formatCurrency(p.budget)}</TableCell>
   298→                      <TableCell className="text-xs">{p.manager?.name || '—'}</TableCell>
   299→                      <TableCell className="text-xs">{formatDate(p.endDate)}</TableCell>
   300→                      <TableCell className="text-right" onClick={(e) => e.stopPropagation()}>
   301→                        <div className="flex items-center justify-end gap-1">
   302→                          <Button size="icon" variant="ghost" className="h-8 w-8" onClick={() => openProject(p.id)} title="View">
   303→                            <ChevronRight className="h-4 w-4" />
   304→                          </Button>
   305→                          {canEdit && (
   306→                            <Button size="icon" variant="ghost" className="h-8 w-8" onClick={() => openEdit(p)} title="Edit">
   307→                              <Pencil className="h-4 w-4" />
   308→                            </Button>
   309→                          )}
   310→                          {canDelete && (
   311→                            <Button
   312→                              size="icon"
   313→                              variant="ghost"
   314→                              className="h-8 w-8 text-red-600 hover:text-red-700 hover:bg-red-50"
   315→                              onClick={() => setDeleteId(p.id)}
   316→                              title="Delete"
   317→                            >
   318→                              <Trash2 className="h-4 w-4" />
   319→                            </Button>
   320→                          )}
   321→                        </div>
   322→                      </TableCell>
   323→                    </TableRow>
   324→                  ))}
   325→                </TableBody>
   326→              </Table>
   327→            </ScrollArea>
   328→          </CardContent>
   329→        </Card>
   330→      )}
   331→
   332→      {/* Create / Edit dialog */}
   333→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   334→        <DialogContent className="sm:max-w-[640px] max-h-[90vh] overflow-y-auto ayk-scrollbar">
   335→          <DialogHeader>
   336→            <DialogTitle>{editId ? 'Edit Project' : 'New Project'}</DialogTitle>
   337→            <DialogDescription>
   338→              {editId ? 'Update the project details below.' : 'Create a new solar construction project.'}
   339→            </DialogDescription>
   340→          </DialogHeader>
   341→          <ProjectFormFields form={form} setForm={setForm} users={users} />
   342→          <DialogFooter>
   343→            <Button variant="outline" onClick={() => setDialogOpen(false)}>Cancel</Button>
   344→            <Button
   345→              onClick={onSubmit}
   346→              disabled={submitting}
   347→              className="bg-emerald-600 hover:bg-emerald-700 text-white"
   348→            >
   349→              {submitting ? 'Saving...' : editId ? 'Save Changes' : 'Create Project'}
   350→            </Button>
   351→          </DialogFooter>
   352→        </DialogContent>
   353→      </Dialog>
   354→
   355→      {/* Delete confirm */}
   356→      <AlertDialog open={!!deleteId} onOpenChange={(o) => !o && setDeleteId(null)}>
   357→        <AlertDialogContent>
   358→          <AlertDialogHeader>
   359→            <AlertDialogTitle>Delete project?</AlertDialogTitle>
   360→            <AlertDialogDescription>
   361→              This will permanently delete the project and all related stages, progress, tasks,
   362→              expenses and documents. This action cannot be undone.
   363→            </AlertDialogDescription>
   364→          </AlertDialogHeader>
   365→          <AlertDialogFooter>
   366→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   367→            <AlertDialogAction
   368→              className="bg-red-600 hover:bg-red-700 text-white"
   369→              onClick={onDelete}
   370→            >
   371→              Delete
   372→            </AlertDialogAction>
   373→          </AlertDialogFooter>
   374→        </AlertDialogContent>
   375→      </AlertDialog>
   376→    </div>
   377→  )
   378→}

   380→function ProjectCard({ project, onOpen }: { project: any; onOpen: () => void }) {
   381→  const installPct =
   382→    project.totalPanels > 0
   383→      ? Math.round((project.installedPanels / project.totalPanels) * 1000) / 10
   384→      : 0
   385→  const budgetPct =
   386→    project.budget > 0 ? Math.min(100, (project.actualCost / project.budget) * 100) : 0
   387→
   388→  return (
   389→    <Card
   390→      className="overflow-hidden hover:shadow-md transition-shadow cursor-pointer group border-border/60"
   391→      onClick={onOpen}
   392→    >
   393→      <CardContent className="p-5">
   394→        <div className="flex items-start justify-between gap-2 mb-3">
   395→          <div className="min-w-0">
   396→            <div className="font-mono text-[10px] text-muted-foreground mb-0.5">{project.code}</div>
   397→            <h3 className="font-semibold text-slate-900 truncate group-hover:text-emerald-600 transition">
   398→              {project.name}
   399→            </h3>
   400→            <div className="flex items-center gap-1 text-xs text-slate-500 mt-0.5">
   401→              <MapPin className="h-3 w-3" /> {project.location}
   402→            </div>
   403→          </div>
   404→          <StatusBadge status={project.status} />
   405→        </div>
   406→
   407→        <div className="space-y-3">
   408→          <div>
   409→            <div className="flex items-center justify-between text-xs mb-1">
   410→              <span className="text-slate-600">Overall Progress</span>
   411→              <span className="font-semibold text-slate-900">{project.overallProgress || 0}%</span>
   412→            </div>
   413→            <Progress value={project.overallProgress || 0} className="h-2" />
   414→          </div>
   415→
   416→          <div className="grid grid-cols-2 gap-3 pt-2 border-t border-slate-100">
   417→            <div>
   418→              <div className="text-[10px] text-slate-500 uppercase tracking-wide">Panels</div>
   419→              <div className="text-sm font-semibold text-slate-900">
   420→                {formatNumber(project.installedPanels)} / {formatNumber(project.totalPanels)}
   421→              </div>
   422→              <Progress value={installPct} className="h-1.5 mt-1" />
   423→            </div>
   424→            <div>
   425→              <div className="text-[10px] text-slate-500 uppercase tracking-wide">Budget</div>
   426→              <div className="text-sm font-semibold text-slate-900">
   427→                {formatCurrency(project.actualCost)} / {formatCurrency(project.budget)}
   428→              </div>
   429→              <Progress value={budgetPct} className="h-1.5 mt-1" />
   430→            </div>
   431→          </div>
   432→
   433→          <div className="flex items-center justify-between text-xs text-slate-500 pt-2 border-t border-slate-100">
   434→            <span className="flex items-center gap-1"><Calendar className="h-3 w-3" /> {formatDate(project.startDate)}</span>
   435→            <span>→</span>
   436→            <span className="flex items-center gap-1">{formatDate(project.endDate)}</span>
   437→          </div>
   438→        </div>
   439→      </CardContent>
   440→    </Card>
   441→  )
   442→}

  1066→// ============================================================
  1067→function StagesCard({
  1068→  project, canEdit, onSaved,
  1069→}: {
  1070→  project: any
  1071→  canEdit: boolean
  1072→  onSaved: () => void
  1073→}) {
  1074→  const [stages, setStages] = useState<any[]>([])
  1075→  const [saving, setSaving] = useState(false)
  1076→
  1077→  useEffect(() => {
  1078→    if (project.stages) {
  1079→      setStages(project.stages.map((s: any) => ({ ...s })))
  1080→    }
  1081→  }, [project.stages])
  1082→
  1083→  function updateStage(idx: number, field: string, value: any) {
  1084→    setStages((prev) =>
  1085→      prev.map((s, i) => (i === idx ? { ...s, [field]: value } : s))
  1086→    )
  1087→  }
  1088→
  1089→  async function save() {
  1090→    setSaving(true)
  1091→    try {
  1092→      await apiPut(`/api/projects/${project.id}/stages`, {
  1093→        stages: stages.map((s) => ({
  1094→          id: s.id,
  1095→          name: s.name,
  1096→          plannedPct: Number(s.plannedPct) || 0,
  1097→          actualPct: Number(s.actualPct) || 0,
  1098→          weight: Number(s.weight) || 0,
  1099→          status: s.status,
  1100→          startDate: s.startDate,
  1101→          endDate: s.endDate,
  1102→        })),
  1103→      })
  1104→      toast.success('Stages saved')
  1105→      onSaved()
  1106→    } catch (e: any) {
  1107→      toast.error(e.message || 'Failed to save stages')
  1108→    } finally {
  1109→      setSaving(false)
  1110→    }
  1111→  }
  1112→
  1113→  return (
  1114→    <Card className="border-border/60 shadow-sm">
  1115→      <CardHeader className="pb-3 flex flex-row items-center justify-between gap-3">
  1116→        <div>
  1117→          <CardTitle className="text-base font-semibold">Project Stages</CardTitle>
  1118→          <CardDescription className="text-xs">Weighted progress across {stages.length || 6} stages</CardDescription>
  1119→        </div>
  1120→        {canEdit && (
  1121→          <Button
  1122→            size="sm"
  1123→            onClick={save}
  1124→            disabled={saving}
  1125→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
  1126→          >
  1127→            {saving ? 'Saving...' : 'Save Stages'}
  1128→          </Button>
  1129→        )}
  1130→      </CardHeader>
  1131→      <CardContent>
  1132→        {stages.length === 0 ? (
  1133→          <EmptyState
  1134→            icon={<FolderKanban className="h-5 w-5" />}
  1135→            title="No stages defined"
  1136→            desc="Stages are created when the project is added."
  1137→          />
  1138→        ) : (
  1139→          <div className="overflow-x-auto">
  1140→            <Table>
  1141→              <TableHeader>
  1142→                <TableRow>
  1143→                  <TableHead>Stage</TableHead>
  1144→                  <TableHead>Planned %</TableHead>
  1145→                  <TableHead>Actual %</TableHead>
  1146→                  <TableHead className="min-w-[140px]">Progress</TableHead>
  1147→                  <TableHead>Weight</TableHead>
  1148→                  <TableHead>Status</TableHead>
  1149→                  <TableHead>Start</TableHead>
  1150→                  <TableHead>End</TableHead>
  1151→                </TableRow>
  1152→              </TableHeader>
  1153→              <TableBody>
  1154→                {stages.map((s, i) => (
  1155→                  <TableRow key={s.id}>
  1156→                    <TableCell className="font-medium text-sm">{s.name}</TableCell>
  1157→                    <TableCell className="text-xs">{s.plannedPct}%</TableCell>
  1158→                    <TableCell>
  1159→                      {canEdit ? (
  1160→                        <Input
  1161→                          type="number"
  1162→                          min={0}
  1163→                          max={100}
  1164→                          value={s.actualPct}
  1165→                          onChange={(e) => updateStage(i, 'actualPct', e.target.value)}
  1166→                          className="h-8 w-20 text-xs"
  1167→                        />
  1168→                      ) : (
  1169→                        <span className="text-xs">{s.actualPct}%</span>
  1170→                      )}
  1171→                    </TableCell>
  1172→                    <TableCell>
  1173→                      <Progress value={Number(s.actualPct) || 0} className="h-2" />
  1174→                    </TableCell>
  1175→                    <TableCell className="text-xs">{s.weight}</TableCell>
  1176→                    <TableCell>
  1177→                      {canEdit ? (
  1178→                        <Select
  1179→                          value={s.status}
  1180→                          onValueChange={(v) => updateStage(i, 'status', v)}
  1181→                        >
  1182→                          <SelectTrigger className="h-8 w-[130px] text-xs">
  1183→                            <SelectValue />
  1184→                          </SelectTrigger>
  1185→                          <SelectContent>

  1190→                          </SelectContent>

  1192→                      ) : (
  1193→                        <StatusBadge status={s.status} />
  1194→                      )}
  1195→                    </TableCell>
  1196→                    <TableCell className="text-xs">{s.startDate ? formatDate(s.startDate) : '—'}</TableCell>
  1197→                    <TableCell className="text-xs">{s.endDate ? formatDate(s.endDate) : '—'}</TableCell>
  1198→                  </TableRow>
  1199→                ))}
  1200→              </TableBody>
  1201→            </Table>
  1202→          </div>
  1203→        )}
  1204→      </CardContent>
  1205→    </Card>
